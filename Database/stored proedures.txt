CREATE PROCEDURE [dbo].[spUTL_PIP_M_CheckListWOPM]
(
	@p_CompanyID VARCHAR(10), @p_LocationID VARCHAR(3),
	@p_dt_Template AS [Type_N_Tbl_C5] READONLY,
	@p_dt_Version AS [Type_N_Tbl_C13] READONLY,
	@p_dt_Section AS [Type_N_Tbl_C5] READONLY,
	@p_dt_SubSection AS [Type_N_Tbl_C5] READONLY,
	@p_dt_Item AS [Type_N_Tbl_C25] READONLY,
	@p_dt_Group AS [Type_Tbl_C7] READONLY,
	@p_dt_RepeatConfig AS [Type_N_Tbl_C13] READONLY,
	@p_dt_RepeatField AS [Type_N_Tbl_C13] READONLY,
	@p_dt_DelTemplate AS [Type_Tbl_C2] READONLY, 
	@p_dt_DelVersion AS [Type_Tbl_C2] READONLY,
	@p_UpdateUser VARCHAR(50),
	@p_IsPublished BIT = 0  
)
AS
SET NOCOUNT ON

DECLARE @tbl_Template TABLE 
(IDTemplate INT, RuntimeTemplateID INT, TemplateNo VARCHAR(50), TemplateName VARCHAR(200), SectionID VARCHAR(10))
INSERT @tbl_Template SELECT * FROM @p_dt_Template

DECLARE @tbl_Version TABLE 
(IDVersion INT, RuntimeVersionID INT, RuntimeTemplateID INT, [Version] VARCHAR(10), Remark VARCHAR(255), [Status] VARCHAR(10), EffectiveDate SMALLDATETIME, StatusOption VARCHAR(50), GroupEquipment VARCHAR(10), SummarySampleText VARCHAR(500), RemarkRequiredOptions VARCHAR(8000), IsRemark1Required BIT, Remark1Text VARCHAR(100), IsRemark2Required BIT, Remark2Text VARCHAR(100), PhotoRequirement VARCHAR(10))
INSERT @tbl_Version SELECT C1, C2, C3, C4, C5, C6, CAST(C7 AS SMALLDATETIME), C8, C9, C10, C11,
 CASE WHEN LEFT(ISNULL(C12,''),2)='1|' THEN 1 ELSE 0 END,
 CASE WHEN LEFT(ISNULL(C12,''),2)='1|' AND CHARINDEX('|',ISNULL(C12,''))>0 THEN SUBSTRING(C12,3,100) ELSE '' END,
 CASE WHEN LEFT(ISNULL(C13,''),2)='1|' THEN 1 ELSE 0 END,
 CASE WHEN LEFT(ISNULL(C13,''),2)='1|' AND CHARINDEX('|',ISNULL(C13,''))>0 THEN SUBSTRING(C13,3,CASE WHEN CHARINDEX('|',C13+'|',3)>0 THEN CHARINDEX('|',C13+'|',3)-3 ELSE 100 END) ELSE '' END,
 CASE WHEN CHARINDEX('|',ISNULL(C13,''),3)>0 AND UPPER(SUBSTRING(C13,CHARINDEX('|',C13,3)+1,10)) IN ('NONE','BEFORE','AFTER','BOTH') THEN UPPER(SUBSTRING(C13,CHARINDEX('|',C13,3)+1,10)) ELSE 'NONE' END
 FROM @p_dt_Version

DECLARE @tbl_Section TABLE 
(IDSection INT, RuntimeSectionID INT, RuntimeVersionID INT, SectionName VARCHAR(200), DisplayOrder INT)
INSERT @tbl_Section SELECT C1, C2, C3, C4, ROW_NUMBER() OVER ( PARTITION BY C3 ORDER BY (SELECT NULL)) FROM @p_dt_Section

DECLARE @tbl_SubSection TABLE
(IDSubSection INT, RuntimeSubSectionID INT, RuntimeSectionID INT, SubSectionName VARCHAR(200), DisplayOrder INT)
INSERT @tbl_SubSection SELECT C1, C2, C3, C4, C5 FROM @p_dt_SubSection

DECLARE @tbl_Item TABLE 
(IDItem INT, RuntimeItemID INT, RuntimeSectionID INT, ItemLabel VARCHAR(500), ItemType VARCHAR(20), Unit VARCHAR(20), IsRemark1Required BIT, IsRemark2Required BIT, IsValueRequired BIT, ShowUnitInRange BIT, OptionList VARCHAR(200), MinValue DECIMAL, MaxValue DECIMAL, DisplayOrder INT, MinTolerance MONEY, MaxTolerance MONEY, ModelCode VARCHAR(50), IsToleranceRequired BIT, RemarkRequiredOptions VARCHAR(8000), Remark1Text VARCHAR(100), Remark2Text VARCHAR(100), RuntimeSubSectionID INT, IsUseDefaultRemark1 BIT, IsUseDefaultRemark2 BIT, IsRemark1Mandatory BIT, IsRemark2Mandatory BIT)
INSERT @tbl_Item SELECT C1, C2, C3, C4, C5, C6, C7, C18, C8, C9, C10, C11, C12, ISNULL(NULLIF(C22, 0), ROW_NUMBER() OVER ( PARTITION BY C3 ORDER BY (SELECT NULL))), C13, C14, C15, CAST(C16 AS BIT), C17, CASE WHEN CAST(ISNULL(NULLIF(C7, ''), '0') AS BIT)=1 THEN ISNULL(NULLIF(C19, ''), 'Remark') ELSE '' END, CASE WHEN CAST(ISNULL(NULLIF(C18, ''), '0') AS BIT)=1 THEN ISNULL(NULLIF(C20, ''), 'Remark') ELSE '' END, ISNULL(C21, 0), CAST(ISNULL(NULLIF(C23,''),'0') AS BIT), CAST(ISNULL(NULLIF(C24,''),'0') AS BIT), CAST(ISNULL(NULLIF(PARSENAME(REPLACE(ISNULL(C25,'0|0'),'|','.'), CASE WHEN LEN(ISNULL(C25,''))-LEN(REPLACE(ISNULL(C25,''),'|',''))>=2 THEN 3 ELSE 2 END),''),'0') AS BIT), CAST(ISNULL(NULLIF(PARSENAME(REPLACE(ISNULL(C25,'0|0'),'|','.'), CASE WHEN LEN(ISNULL(C25,''))-LEN(REPLACE(ISNULL(C25,''),'|',''))>=2 THEN 2 ELSE 1 END),''),'0') AS BIT) FROM @p_dt_Item

DECLARE @tbl_Group TABLE
(IDGroup INT, RuntimeGroupID INT, RuntimeItemID INT, GroupName VARCHAR(100), FieldList VARCHAR(100), Unit VARCHAR(20), DisplayOrder INT)
INSERT @tbl_Group SELECT C1, C2, C3, C4, C5, C6, ROW_NUMBER() OVER ( PARTITION BY C3 ORDER BY (SELECT NULL)) FROM @p_dt_Group

DECLARE @tbl_RepeatConfig TABLE
(IDItem INT, RuntimeItemID INT, RepeatMode VARCHAR(10), ColumnCount INT, ObjectLabel NVARCHAR(100),
 CountMode VARCHAR(10), FixedCount INT NULL, MinCount INT NULL, MaxCount INT NULL)
INSERT @tbl_RepeatConfig
SELECT CAST(C1 AS INT), CAST(C2 AS INT), UPPER(C3),
       ISNULL(NULLIF(C4,''), '3'), NULLIF(C5,''),
       UPPER(ISNULL(NULLIF(C6,''),'FLEX')),
       NULLIF(C7,''), NULLIF(C8,''), NULLIF(C9,'')
FROM @p_dt_RepeatConfig;

DECLARE @tbl_RepeatField TABLE
(RuntimeItemID INT, FieldKey VARCHAR(50), FieldLabel NVARCHAR(100), FieldType VARCHAR(20),
 OptionList NVARCHAR(1000), Unit NVARCHAR(30), MinValue DECIMAL(18,4) NULL,
 MaxValue DECIMAL(18,4) NULL, IsRequired BIT, GroupName NVARCHAR(100), DisplayOrder INT,
 MinInclusive BIT, MaxInclusive BIT)
INSERT @tbl_RepeatField
SELECT CAST(C1 AS INT), LTRIM(RTRIM(C2)), C3, UPPER(C4), NULLIF(C5,''), NULLIF(C6,''),
       NULLIF(C7,''), NULLIF(C8,''), CAST(ISNULL(NULLIF(C9,''),'0') AS BIT),
       NULLIF(C10,''), CAST(ISNULL(NULLIF(C11,''),'0') AS INT),
       CAST(ISNULL(NULLIF(C12,''),'1') AS BIT), CAST(ISNULL(NULLIF(C13,''),'1') AS BIT)
FROM @p_dt_RepeatField;

IF EXISTS
(
    SELECT 1 FROM @tbl_RepeatField F
    WHERE (F.MinValue IS NOT NULL AND F.MaxValue IS NOT NULL AND F.MinValue>F.MaxValue)
       OR (F.MinValue IS NOT NULL AND F.MaxValue IS NOT NULL AND F.MinValue=F.MaxValue
           AND (F.MinInclusive=0 OR F.MaxInclusive=0))
)
BEGIN RAISERROR('Invalid REPEAT field range boundaries.',16,1); RETURN; END;

IF EXISTS
(
    SELECT 1 FROM @tbl_RepeatConfig C
    WHERE C.RepeatMode NOT IN ('FIELDS','OBJECTS')
       OR C.ColumnCount NOT BETWEEN 1 AND 6
       OR C.CountMode NOT IN ('FIXED','FLEX')
       OR (C.CountMode='FIXED' AND ISNULL(C.FixedCount,0)<1)
       OR (C.CountMode='FLEX' AND C.FixedCount IS NOT NULL)
       OR (C.CountMode='FLEX' AND C.MinCount IS NOT NULL AND C.MaxCount IS NOT NULL AND C.MaxCount<C.MinCount)
       OR NOT EXISTS (SELECT 1 FROM @tbl_Item I WHERE I.RuntimeItemID=C.RuntimeItemID AND I.ItemType='REPEAT')
)
BEGIN RAISERROR('Invalid REPEAT configuration.',16,1); RETURN; END;

IF EXISTS
(
    SELECT 1 FROM @tbl_Item I
    WHERE I.ItemType='REPEAT'
      AND NOT EXISTS (SELECT 1 FROM @tbl_RepeatConfig C WHERE C.RuntimeItemID=I.RuntimeItemID)
)
BEGIN RAISERROR('Every REPEAT item requires a REPEAT configuration.',16,1); RETURN; END;

IF EXISTS
(
    SELECT 1 FROM @tbl_RepeatConfig C
    WHERE NOT EXISTS (SELECT 1 FROM @tbl_RepeatField F WHERE F.RuntimeItemID=C.RuntimeItemID)
)
BEGIN RAISERROR('Every REPEAT item requires at least one child field.',16,1); RETURN; END;

IF EXISTS
(
    SELECT 1 FROM @tbl_RepeatField F
    WHERE F.FieldKey='' OR LEN(F.FieldKey)>50
       OR F.FieldType NOT IN ('NUMBER','RANGE','OPTION','TEXT')
       OR NOT EXISTS (SELECT 1 FROM @tbl_RepeatConfig C WHERE C.RuntimeItemID=F.RuntimeItemID)
)
BEGIN RAISERROR('Invalid REPEAT field definition. Nested REPEAT fields are not supported.',16,1); RETURN; END;

DECLARE @tbl_DelTemplate TABLE (IDTemplate INT)
INSERT @tbl_DelTemplate SELECT C1 FROM @p_dt_DelTemplate

DECLARE @tbl_DelVersion TABLE (IDTemplate INT, IDVersion INT)
INSERT @tbl_DelVersion SELECT * FROM @p_dt_DelVersion

DECLARE @tbl_Identity TABLE (Entity VARCHAR(50), RuntimeID INT, ID INT)

DECLARE @p_IsDebug BIT = 0
IF @p_IsDebug=1
BEGIN
	IF EXISTS (SELECT * FROM INFORMATION_SCHEMA.TABLES AS t WHERE t.TABLE_NAME='tmpDebug_PIP_M_CheckListWOPM')
		DROP TABLE tmpDebug_PIP_M_CheckListWOPM
	IF EXISTS (SELECT * FROM INFORMATION_SCHEMA.TABLES AS t WHERE t.TABLE_NAME='tmpDebug_PIP_M_CheckListWOPM_Version')
		DROP TABLE tmpDebug_PIP_M_CheckListWOPM_Version
	IF EXISTS (SELECT * FROM INFORMATION_SCHEMA.TABLES AS t WHERE t.TABLE_NAME='tmpDebug_PIP_M_CheckListWOPM_Section')
		DROP TABLE tmpDebug_PIP_M_CheckListWOPM_Section
	IF EXISTS (SELECT * FROM INFORMATION_SCHEMA.TABLES AS t WHERE t.TABLE_NAME='tmpDebug_PIP_M_CheckListWOPM_Item')
		DROP TABLE tmpDebug_PIP_M_CheckListWOPM_Item
	IF EXISTS (SELECT * FROM INFORMATION_SCHEMA.TABLES AS t WHERE t.TABLE_NAME='tmpDebug_PIP_M_CheckListWOPM_ItemGroup')
		DROP TABLE tmpDebug_PIP_M_CheckListWOPM_ItemGroup
	IF EXISTS (SELECT * FROM INFORMATION_SCHEMA.TABLES AS t WHERE t.TABLE_NAME='tmpDebug_PIP_M_CheckListWOPM_SubSection')
		DROP TABLE tmpDebug_PIP_M_CheckListWOPM_SubSection
	IF EXISTS (SELECT * FROM INFORMATION_SCHEMA.TABLES AS t WHERE t.TABLE_NAME='tmpDebug_PIP_M_CheckListWOPM_SubSectionSource')

	SELECT * INTO tmpDebug_PIP_M_CheckListWOPM FROM @tbl_Template

	SELECT * INTO tmpDebug_PIP_M_CheckListWOPM_Item FROM @tbl_Item
	SELECT * INTO tmpDebug_PIP_M_CheckListWOPM_ItemGroup FROM @tbl_Group
	SELECT * INTO tmpDebug_PIP_M_CheckListWOPM_SubSection FROM @tbl_SubSection
END


/* Published/Active and archived versions are immutable at the database boundary. */
IF EXISTS (SELECT 1 FROM @tbl_Version S JOIN dbo.PIP_M_CheckListWOPM_Version V ON V.IDVersion=S.IDVersion
           WHERE S.IDVersion>0 AND ISNULL(V.[Status],'')<>'Draft')
BEGIN RAISERROR('Only Draft versions can be edited or published.',16,1); RETURN; END;
IF EXISTS (SELECT 1 FROM @tbl_DelVersion D JOIN dbo.PIP_M_CheckListWOPM_Version V ON V.IDVersion=D.IDVersion
           WHERE ISNULL(V.[Status],'')<>'Draft')
BEGIN RAISERROR('Only Draft versions can be deleted.',16,1); RETURN; END;
IF EXISTS (SELECT 1 FROM @tbl_Section X JOIN dbo.PIP_M_CheckListWOPM_Section S ON S.IDSection=X.IDSection
           JOIN dbo.PIP_M_CheckListWOPM_Version V ON V.IDVersion=S.IDVersion
           WHERE X.IDSection>0 AND ISNULL(V.[Status],'')<>'Draft')
OR EXISTS (SELECT 1 FROM @tbl_Item X JOIN dbo.PIP_M_CheckListWOPM_Item I ON I.IDItem=X.IDItem
           JOIN dbo.PIP_M_CheckListWOPM_Section S ON S.IDSection=I.IDSection
           JOIN dbo.PIP_M_CheckListWOPM_Version V ON V.IDVersion=S.IDVersion
           WHERE X.IDItem>0 AND ISNULL(V.[Status],'')<>'Draft')
OR EXISTS (SELECT 1 FROM @tbl_Group X JOIN dbo.PIP_M_CheckListWOPM_ItemGroup G ON G.IDGroup=X.IDGroup
           JOIN dbo.PIP_M_CheckListWOPM_Item I ON I.IDItem=G.IDItem
           JOIN dbo.PIP_M_CheckListWOPM_Section S ON S.IDSection=I.IDSection
           JOIN dbo.PIP_M_CheckListWOPM_Version V ON V.IDVersion=S.IDVersion
           WHERE X.IDGroup>0 AND ISNULL(V.[Status],'')<>'Draft')
OR EXISTS (SELECT 1 FROM @tbl_SubSection X JOIN dbo.PIP_M_CheckListWOPM_SubSection SS ON SS.IDSubSection=X.IDSubSection
           JOIN dbo.PIP_M_CheckListWOPM_Section S ON S.IDSection=SS.IDSection
           JOIN dbo.PIP_M_CheckListWOPM_Version V ON V.IDVersion=S.IDVersion
           WHERE X.IDSubSection>0 AND ISNULL(V.[Status],'')<>'Draft')
BEGIN RAISERROR('Published/Active version structure is immutable.',16,1); RETURN; END;
IF EXISTS (SELECT 1 FROM @tbl_DelTemplate D JOIN dbo.PIP_M_CheckListWOPM_Version V ON V.IDTemplate=D.IDTemplate
           WHERE V.[Status] IN ('Active','Archieved'))
BEGIN RAISERROR('Templates with published versions cannot be deleted.',16,1); RETURN; END;

--- begin processsing
--- DELETED
DELETE S2 FROM PIP_M_CheckListWOPM_itemgroup S2 JOIN PIP_M_CheckListWOPM_Item S1 ON S1.IDItem = S2.IDItem
JOIN PIP_M_CheckListWOPM_Section S3 ON S3.IDSection = S1.IDSection
JOIN @tbl_Version S4 ON S4.IDVersion=S3.IDVersion
JOIN PIP_M_CheckListWOPM_Version AS pmclwv ON pmclwv.IDVersion = S4.IDVersion
JOIN PIP_M_CheckListWOPM AS pmclw ON pmclw.IDTemplate=pmclwv.IDTemplate
WHERE NOT EXISTS (SELECT * FROM @tbl_Group S WHERE S.IDGroup=S2.IDGroup)
AND (ISNULL(pmclw.CompanyScope,'S')='A'
     OR (';' + ISNULL(pmclw.CompanyLocationList, '') + ';' LIKE '%;' + @p_CompanyID + @p_LocationID + ';%'))


DELETE RF
FROM dbo.PIP_M_CheckListWOPM_ItemRepeatField RF
JOIN dbo.PIP_M_CheckListWOPM_Item RI ON RI.IDItem=RF.IDItem
JOIN dbo.PIP_M_CheckListWOPM_Section RS ON RS.IDSection=RI.IDSection
JOIN @tbl_Version RV ON RV.IDVersion=RS.IDVersion
WHERE NOT EXISTS (SELECT 1 FROM @tbl_Item KeepItem WHERE KeepItem.IDItem=RI.IDItem);
DELETE R
FROM dbo.PIP_M_CheckListWOPM_ItemRepeat R
JOIN dbo.PIP_M_CheckListWOPM_Item RI ON RI.IDItem=R.IDItem
JOIN dbo.PIP_M_CheckListWOPM_Section RS ON RS.IDSection=RI.IDSection
JOIN @tbl_Version RV ON RV.IDVersion=RS.IDVersion
WHERE NOT EXISTS (SELECT 1 FROM @tbl_Item KeepItem WHERE KeepItem.IDItem=RI.IDItem);

DELETE S2 FROM PIP_M_CheckListWOPM_item S2 JOIN PIP_M_CheckListWOPM_Section S1 ON S1.IDSection = S2.IDSection
JOIN @tbl_Version S3 ON S3.IDVersion=S1.IDVersion
JOIN PIP_M_CheckListWOPM_Version AS pmclwv ON pmclwv.IDVersion = S3.IDVersion
JOIN PIP_M_CheckListWOPM AS pmclw ON pmclw.IDTemplate=pmclwv.IDTemplate
WHERE NOT EXISTS (SELECT * FROM @tbl_Item S WHERE S.IDItem=S2.IDItem)
AND (ISNULL(pmclw.CompanyScope,'S')='A'
     OR (';' + ISNULL(pmclw.CompanyLocationList, '') + ';' LIKE '%;' + @p_CompanyID + @p_LocationID + ';%'))


DELETE OldSub
FROM PIP_M_CheckListWOPM_SubSection OldSub
JOIN PIP_M_CheckListWOPM_Section OldSec ON OldSec.IDSection=OldSub.IDSection
JOIN @tbl_Version V ON V.IDVersion=OldSec.IDVersion
JOIN PIP_M_CheckListWOPM_Version VER ON VER.IDVersion=V.IDVersion
JOIN PIP_M_CheckListWOPM T ON T.IDTemplate=VER.IDTemplate
WHERE NOT EXISTS (SELECT 1 FROM @tbl_Section S WHERE S.IDSection=OldSec.IDSection)
AND (ISNULL(T.CompanyScope,'S')='A' OR (';' + ISNULL(T.CompanyLocationList, '') + ';' LIKE '%;' + @p_CompanyID + @p_LocationID + ';%'));

DELETE S2 FROM PIP_M_CheckListWOPM_section S2 JOIN @tbl_Version S1 ON S1.IDVersion = S2.IDVersion
JOIN PIP_M_CheckListWOPM_Version AS pmclwv ON pmclwv.IDVersion = S2.IDVersion
JOIN PIP_M_CheckListWOPM AS pmclw ON pmclw.IDTemplate=pmclwv.IDTemplate
WHERE NOT EXISTS (SELECT * FROM @tbl_Section S WHERE S.IDSection=S2.IDSection)
AND (ISNULL(pmclw.CompanyScope,'S')='A'
     OR (';' + ISNULL(pmclw.CompanyLocationList, '') + ';' LIKE '%;' + @p_CompanyID + @p_LocationID + ';%'))


IF EXISTS (SELECT * FROM @tbl_DelVersion)
BEGIN
	DELETE A FROM PIP_M_CheckListWOPM_ItemGroup A JOIN PIP_M_CheckListWOPM_item AA ON AA.IDItem = A.IDItem
	JOIN PIP_M_CheckListWOPM_section AAA ON AAA.IDSection = AA.IDSection
	JOIN @tbl_DelVersion B ON B.IDVersion=AAA.IDVersion
	JOIN PIP_M_CheckListWOPM_Version AS pmclwv ON pmclwv.IDVersion = B.IDVersion
	JOIN PIP_M_CheckListWOPM AS pmclw ON pmclw.IDTemplate=pmclwv.IDTemplate
	WHERE (ISNULL(pmclw.CompanyScope,'S')='A'
       OR (';' + ISNULL(pmclw.CompanyLocationList, '') + ';' LIKE '%;' + @p_CompanyID + @p_LocationID + ';%'))
	
	DELETE RF FROM dbo.PIP_M_CheckListWOPM_ItemRepeatField RF
	JOIN dbo.PIP_M_CheckListWOPM_Item RI ON RI.IDItem=RF.IDItem
	JOIN dbo.PIP_M_CheckListWOPM_Section RS ON RS.IDSection=RI.IDSection
	JOIN @tbl_DelVersion DV ON DV.IDVersion=RS.IDVersion;
	DELETE R FROM dbo.PIP_M_CheckListWOPM_ItemRepeat R
	JOIN dbo.PIP_M_CheckListWOPM_Item RI ON RI.IDItem=R.IDItem
	JOIN dbo.PIP_M_CheckListWOPM_Section RS ON RS.IDSection=RI.IDSection
	JOIN @tbl_DelVersion DV ON DV.IDVersion=RS.IDVersion;

	DELETE A FROM PIP_M_CheckListWOPM_item A JOIN PIP_M_CheckListWOPM_section AA ON AA.IDSection = A.IDSection
	JOIN @tbl_DelVersion B ON B.IDVersion=AA.IDVersion
	JOIN PIP_M_CheckListWOPM_Version AS pmclwv ON pmclwv.IDVersion = B.IDVersion
	JOIN PIP_M_CheckListWOPM AS pmclw ON pmclw.IDTemplate=pmclwv.IDTemplate
	WHERE (ISNULL(pmclw.CompanyScope,'S')='A'
       OR (';' + ISNULL(pmclw.CompanyLocationList, '') + ';' LIKE '%;' + @p_CompanyID + @p_LocationID + ';%'))
	
	DELETE SS FROM PIP_M_CheckListWOPM_SubSection SS
	JOIN PIP_M_CheckListWOPM_section AAA ON AAA.IDSection=SS.IDSection
	JOIN @tbl_DelVersion B ON B.IDVersion=AAA.IDVersion
	JOIN PIP_M_CheckListWOPM_Version VER ON VER.IDVersion=B.IDVersion
	JOIN PIP_M_CheckListWOPM T ON T.IDTemplate=VER.IDTemplate
	WHERE (ISNULL(T.CompanyScope,'S')='A' OR (';' + ISNULL(T.CompanyLocationList, '') + ';' LIKE '%;' + @p_CompanyID + @p_LocationID + ';%'));

	DELETE A FROM PIP_M_CheckListWOPM_section A JOIN @tbl_DelVersion B ON B.IDVersion=A.IDVersion
	JOIN PIP_M_CheckListWOPM_Version AS pmclwv ON pmclwv.IDVersion = B.IDVersion
	JOIN PIP_M_CheckListWOPM AS pmclw ON pmclw.IDTemplate=pmclwv.IDTemplate
	WHERE (ISNULL(pmclw.CompanyScope,'S')='A'
       OR (';' + ISNULL(pmclw.CompanyLocationList, '') + ';' LIKE '%;' + @p_CompanyID + @p_LocationID + ';%'))
	
END

DELETE V FROM PIP_M_CheckListWOPM_version V
INNER JOIN @tbl_DelVersion D ON V.IDTemplate = D.IDTemplate AND V.IDVersion = D.IDVersion
JOIN PIP_M_CheckListWOPM AS pmclw ON pmclw.IDTemplate=V.IDTemplate
WHERE (ISNULL(pmclw.CompanyScope,'S')='A'
       OR (';' + ISNULL(pmclw.CompanyLocationList, '') + ';' LIKE '%;' + @p_CompanyID + @p_LocationID + ';%'))


/* Resolve which requested templates are visible in this company/location before removing their relations. */
DECLARE @tbl_AllowedDelTemplate TABLE (IDTemplate INT PRIMARY KEY);
INSERT INTO @tbl_AllowedDelTemplate (IDTemplate)
SELECT T.IDTemplate
FROM dbo.PIP_M_CheckListWOPM T
JOIN @tbl_DelTemplate D ON D.IDTemplate=T.IDTemplate
WHERE ISNULL(T.CompanyScope,'S')='A'
   OR (';' + ISNULL(T.CompanyLocationList, '') + ';' LIKE '%;' + @p_CompanyID + @p_LocationID + ';%');

UPDATE I SET I.IDSubSection=NULL
FROM PIP_M_CheckListWOPM_Item I
JOIN PIP_M_CheckListWOPM_Section S ON S.IDSection=I.IDSection
JOIN PIP_M_CheckListWOPM_Version V ON V.IDVersion=S.IDVersion
JOIN @tbl_AllowedDelTemplate D ON D.IDTemplate=V.IDTemplate;

DELETE SS
FROM PIP_M_CheckListWOPM_SubSection SS
JOIN PIP_M_CheckListWOPM_Section S ON S.IDSection=SS.IDSection
JOIN PIP_M_CheckListWOPM_Version V ON V.IDVersion=S.IDVersion
JOIN @tbl_AllowedDelTemplate D ON D.IDTemplate=V.IDTemplate;

DELETE T FROM PIP_M_CheckListWOPM T
JOIN @tbl_AllowedDelTemplate D ON T.IDTemplate = D.IDTemplate


--- TEMPLATE
UPDATE T
SET
  T.TemplateNo = S.TemplateNo,
  T.TemplateName = S.TemplateName,
  T.SectionID = S.SectionID,
  T.UpdateUser=@p_UpdateUser, T.UpdateDate=GETDATE()
FROM PIP_M_CheckListWOPM T
JOIN @tbl_Template S ON T.IDTemplate = S.IDTemplate
WHERE S.IDTemplate > 0 
MERGE INTO PIP_M_CheckListWOPM AS target
USING (
    SELECT 
        templateno, 
        templatename, 
        sectionid, 
        runtimetemplateid,
        -- Menggunakan ISNULL(..., 0) untuk antisipasi jika tabel target masih kosong
        (SELECT ISNULL(MAX(IDTemplate), 0) FROM PIP_M_CheckListWOPM) 
        + ROW_NUMBER() OVER (ORDER BY (SELECT NULL)) AS NewIDTemplate
    FROM @tbl_template 
    WHERE idtemplate = 0
) AS source
ON 1 = 0 
WHEN NOT MATCHED THEN
    INSERT (IDTemplate, TemplateNo, TemplateName, SectionID, IsActive, EntryUser, EntryDate, UpdateUser, UpdateDate, CompanyLocationList, EmployeeList)
    VALUES (source.NewIDTemplate, source.templateno, source.templatename, source.sectionid, 1, @p_UpdateUser, GETDATE(), @p_UpdateUser, GETDATE(), @p_CompanyID + @p_LocationID, '')
OUTPUT 
    'Template' Entity,
    source.runtimetemplateid,   
    inserted.IDTemplate   -- Tetap mengembalikan IDTemplate manual yang baru saja dimasukkan
INTO @tbl_Identity (Entity, RuntimeID, ID);

--- Version
UPDATE V
SET 
    V.Version = S.Version,
    V.[Status] = S.[Status],
    V.Remark = S.Remark,
    V.EffectiveDate = S.EffectiveDate,
    V.StatusOption = S.StatusOption, V.UpdateUser=@p_UpdateUser, V.UpdateDate=GETDATE(), 
    V.GroupEquipment=S.GroupEquipment, V.SummarySampleText=S.SummarySampleText, V.RemarkRequiredOptions=S.RemarkRequiredOptions,
    V.IsRemark1Required=S.IsRemark1Required, V.Remark1Text=CASE WHEN S.IsRemark1Required=1 THEN ISNULL(NULLIF(S.Remark1Text,''),'Remark') ELSE '' END,
    V.IsRemark2Required=S.IsRemark2Required, V.Remark2Text=CASE WHEN S.IsRemark2Required=1 THEN ISNULL(NULLIF(S.Remark2Text,''),'Remark') ELSE '' END,
    V.PhotoRequirement=S.PhotoRequirement
FROM PIP_M_CheckListWOPM_version V
JOIN @tbl_Version S ON V.IDVersion = S.IDVersion
WHERE S.IDVersion > 0 
MERGE INTO PIP_M_CheckListWOPM_version AS target
USING (
    SELECT 
        v.RuntimeVersionID,
        v.RuntimeTemplateID,
        v.[Version],
        v.Remark,
        v.[Status],
        v.EffectiveDate,
        v.StatusOption,
        v.GroupEquipment, v.SummarySampleText, v.RemarkRequiredOptions, v.IsRemark1Required, v.Remark1Text, v.IsRemark2Required, v.Remark2Text, v.PhotoRequirement,
        tmpl.IDTemplate, -- Hubungkan dengan tabel map untuk mendapatkan IDTemplate asli database
        (SELECT ISNULL(MAX(IDVersion), 0) FROM PIP_M_CheckListWOPM_version) 
        + ROW_NUMBER() OVER (ORDER BY (SELECT NULL)) AS NewIDVersion
    FROM @tbl_Version v
    INNER JOIN (
        -- Gabungkan data template baru dan template existing di sini
        SELECT RuntimeID RuntimeTemplateID, ID IDTemplate FROM @tbl_Identity WHERE Entity='Template'  -- Template baru
        UNION ALL
        SELECT RuntimeTemplateID, IDTemplate FROM @tbl_Template WHERE IDTemplate <> 0 -- Template lama
    ) tmpl ON v.RuntimeTemplateID = tmpl.RuntimeTemplateID

CREATE PROCEDURE [dbo].[spUTL_PIP_T_Inspection_Approval]
(
    @p_InspectionID BIGINT,
    @p_Action VARCHAR(10),
    @p_EmployeeID VARCHAR(50),
    @p_CompanyID VARCHAR(10),
    @p_LocationID VARCHAR(3),
    @p_Remark VARCHAR(500) = NULL
)
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT ON;

    SET @p_Action = UPPER(LTRIM(RTRIM(ISNULL(@p_Action, ''))));
    IF @p_Action NOT IN ('APPROVE', 'VOID')
    BEGIN
        RAISERROR('Invalid inspection approval action.', 16, 1);
        RETURN;
    END;
    IF @p_Action = 'VOID' AND LEN(LTRIM(RTRIM(ISNULL(@p_Remark, '')))) = 0
    BEGIN
        RAISERROR('Reason for void is required.', 16, 1);
        RETURN;
    END;
    IF ISNULL(@p_EmployeeID, '') = ''
    BEGIN
        RAISERROR('Supervisor identity is required.', 16, 1);
        RETURN;
    END;

    BEGIN TRY
        BEGIN TRANSACTION;

        DECLARE @V_SectionID VARCHAR(10);
        DECLARE @V_Status VARCHAR(10);
        DECLARE @V_IsApprove BIT;
        DECLARE @V_ApprovedBy VARCHAR(50);
        DECLARE @V_ApprovedDate DATETIME;
        DECLARE @V_ApprovalRemark VARCHAR(500);
        DECLARE @V_ReasonVoid VARCHAR(500);
        DECLARE @V_FollowUp VARCHAR(30);
        DECLARE @V_FollowUpSection VARCHAR(10);
        DECLARE @V_FollowUpRemark VARCHAR(300);
        DECLARE @V_FollowUpWONo VARCHAR(10);
        DECLARE @V_InspectionNo VARCHAR(10);
        DECLARE @V_OwnerEmployeeID VARCHAR(50);
        DECLARE @V_IDTemplate INT;

        SELECT @V_IDTemplate=IDTemplate, @V_SectionID=SectionID, @V_Status=Status, @V_IsApprove=IsApprove,
               @V_ApprovedBy=ApprovedBy, @V_ApprovedDate=ApprovedDate,
               @V_ApprovalRemark=ApprovalRemark, @V_ReasonVoid=ReasonVoid,
               @V_FollowUp=ISNULL(FollowUp,'No Follow Up'),
               @V_FollowUpSection=ISNULL(FollowUpSection,''),
               @V_FollowUpRemark=ISNULL(FollowUpRemark,''),
               @V_FollowUpWONo=ISNULL(FollowUpWONo,''),
               @V_InspectionNo=ISNULL(InspectionNo,''),
               @V_OwnerEmployeeID=OwnerEmployeeID
        FROM dbo.PIP_T_Inspection WITH (UPDLOCK, HOLDLOCK)
        WHERE InspectionID=@p_InspectionID
          AND CompanyID=@p_CompanyID AND LocationID=@p_LocationID;

        IF @V_SectionID IS NULL
            RAISERROR('Inspection was not found for this company/location.', 16, 1);

        IF NOT EXISTS
        (
            SELECT 1
            FROM dbo.UTL_User U
            WHERE U.EmployeeID=@p_EmployeeID
              AND ISNULL(U.IsSPV,0)=1
              AND EXISTS (SELECT 1 FROM dbo.SplitStringConvertToTable(U.Section,';') S WHERE S.val=@V_SectionID)
              AND
              (
                  EXISTS (SELECT 1 FROM dbo.SysWebUserCompanyLocation C
                          WHERE C.UserID=U.UserID AND C.AppID='pip'
                            AND C.CompanyID=@p_CompanyID AND C.LocationID=@p_LocationID)
                  OR EXISTS (SELECT 1 FROM dbo.F_SelectOneLogin('pip', U.EmployeeID, '') L
                             WHERE L.CompanyID=@p_CompanyID AND L.LocationID=@p_LocationID)
              )
        )
        OR NOT EXISTS
        (
            SELECT 1
            FROM dbo.PIP_M_CheckListWOPM T
            WHERE T.IDTemplate=@V_IDTemplate
              AND T.SectionID=@V_SectionID
              AND ISNULL(T.TemplateType,'W')='I'
              AND (ISNULL(T.CompanyScope,'S')='A'
                   OR (';' + ISNULL(T.CompanyLocationList,'') + ';' LIKE '%;' + @p_CompanyID + @p_LocationID + ';%'))
              AND (ISNULL(T.EmployeeList,'')=''
                   OR (';' + ISNULL(T.EmployeeList,'') + ';' LIKE '%;' + @V_OwnerEmployeeID + ';%'))
        )
            RAISERROR('Supervisor is not authorized for this inspection template, technician, or company/location.', 16, 1);

        IF @V_Status <> 'DONE' OR ISNULL(@V_IsApprove,0) <> 0
           OR ISNULL(@V_ApprovedBy,'') <> '' OR ISNULL(@V_ApprovedDate,CONVERT(DATETIME,'19000101',112)) > CONVERT(DATETIME,'19000101',112)
           OR ISNULL(@V_ApprovalRemark,'') <> '' OR ISNULL(@V_ReasonVoid,'') <> ''
            RAISERROR('Only inspections awaiting approval can be processed.', 16, 1);

        IF @p_Action = 'APPROVE'
        BEGIN
            IF @V_FollowUp NOT IN ('No Follow Up','Recommend Follow Up')
                RAISERROR('Inspection Follow Up selection is invalid.', 16, 1);

            IF @V_FollowUp='Recommend Follow Up'
            BEGIN
                IF @V_FollowUpSection NOT IN ('ACON','COMM','MECH','ELEC','CIVIL','PFIT','PHKP','PSEC')
                    RAISERROR('A valid Follow Up Responsible Section is required.', 16, 1);
                IF LEN(LTRIM(RTRIM(@V_FollowUpRemark)))=0
                    RAISERROR('Follow Up Work Order Description is required.', 16, 1);
            END;

            UPDATE dbo.PIP_T_Inspection
            SET IsApprove=1, ApprovedBy=@p_EmployeeID, ApprovedDate=GETDATE(),
                ApprovalRemark=ISNULL(NULLIF(LTRIM(RTRIM(@p_Remark)), ''), ''),
                ReasonVoid='', UpdateUser=@p_EmployeeID, UpdateDate=GETDATE()
            WHERE InspectionID=@p_InspectionID AND CompanyID=@p_CompanyID AND LocationID=@p_LocationID
              AND Status='DONE' AND IsApprove=0 AND ISNULL(ApprovedBy,'')='' AND ISNULL(ApprovedDate,CONVERT(DATETIME,'19000101',112))<=CONVERT(DATETIME,'19000101',112)
              AND ISNULL(ApprovalRemark,'')='' AND ISNULL(ReasonVoid,'')='';

            IF @@ROWCOUNT <> 1
                RAISERROR('Inspection approval state changed before this action completed. Refresh the queue.', 16, 1);

            /* Create the internal WO in the same transaction as approval. The stored WONo
               on the Inspection header prevents a duplicate if the approval is retried. */
            IF @V_FollowUp='Recommend Follow Up' AND ISNULL(@V_FollowUpWONo,'')=''
            BEGIN
                DECLARE @V_KeyHDWO INT;
                DECLARE @V_WONo VARCHAR(10);
                DECLARE @V_AutoNum VARCHAR(30);
                DECLARE @V_RequestedBy VARCHAR(50);
                DECLARE @V_DescriptionWO VARCHAR(255);
                DECLARE @V_AutoNumResult TABLE (ID VARCHAR(10), SeqNo INT);

                SELECT @V_KeyHDWO=ISNULL(MAX(KeyHD),0)+1
                FROM dbo.PIP_T_WO WITH (TABLOCKX, HOLDLOCK);

                SET @V_AutoNum='PIPWO'+@p_CompanyID+@p_LocationID;
                INSERT INTO @V_AutoNumResult
                EXECUTE dbo.sp_GetGenerateAutoNumID 'pip', @V_AutoNum, 1;
                SELECT @V_WONo=ID FROM @V_AutoNumResult;

                IF ISNULL(@V_WONo,'')=''
                    RAISERROR('Failed to generate internal Follow Up WO number.', 16, 1);

                SELECT @V_RequestedBy=LEFT(ISNULL(
                    (SELECT TOP 1 SWU.UserID FROM dbo.SysWebUser SWU WHERE SWU.EmployeeID=@V_OwnerEmployeeID),
                    @V_OwnerEmployeeID),50);

                SET @V_DescriptionWO=LEFT(
                    CASE WHEN @V_InspectionNo='' THEN '' ELSE 'Inspection '+@V_InspectionNo+': ' END+
                    @V_FollowUpRemark,255);

                INSERT INTO dbo.PIP_T_WO
                (
                    KeyHD, CompanyID, LocationID, WONo, WODate, WOType, StartDate, JONo,
                    PMType, Description, LocationNo, SectionID, RequestedBy, CompletedBy, CompletedDate,
                    CancelReason, SIIAmount, IsDelete, IsJO, IsApproved, ApprovedUser, ApprovedDate,
                    EntryUser, EntryDate, UpdateUser, UpdateDate, IsPrint, [RePrint],
                    Feedback1, Feedback2, Feedback3, Feedback4, Feedback5, EquipmentCode, SectionIDUser,
                    isApprovedDH, ApprovedDHUser, ApprovedDHDate, ReasonDH,
                    isApprovedHelpDesk, ApprovedHelpDeskUser, ApprovedHelpDeskDate, ReasonHelpDesk,
                    isCompleteDC, ReasonDC, LastStatusTransaction, ProfitCenterNo,
                    isRespondDC, ApprovedRespondDCUser, ApprovedRespondDCDate, ReasonRespondDC,
                    isCompletedUser, CompletedUser, CompletedDateUser, ReasonUser
                )
                VALUES
                (
                    @V_KeyHDWO, @p_CompanyID, @p_LocationID, @V_WONo,
                    CONVERT(VARCHAR(8),GETDATE(),112), 'I', CONVERT(VARCHAR(8),GETDATE(),112), '',
                    '', @V_DescriptionWO, '', @V_FollowUpSection, @V_RequestedBy, '', '',
                    '', 0, 0, 1, 0, '', '',
                    @p_EmployeeID, GETDATE(), @p_EmployeeID, GETDATE(), 0, 0,
                    0, 0, 0, 0, 0, '', @V_SectionID,
                    1, @p_EmployeeID, GETDATE(), '',
                    1, @p_EmployeeID, GETDATE(), '',
                    0, '', '', '',
                    0, '', '', '',
                    0, '', '', ''
                );

                UPDATE dbo.PIP_T_Inspection
                SET FollowUpWONo=@V_WONo, UpdateUser=@p_EmployeeID, UpdateDate=GETDATE()
                WHERE InspectionID=@p_InspectionID;
            END;
        END
        ELSE
        BEGIN
            UPDATE dbo.PIP_T_Inspection
            SET Status='DRAFT', IsApprove=0, ApprovedBy=@p_EmployeeID, ApprovedDate=GETDATE(),
                ApprovalRemark='', ReasonVoid=LTRIM(RTRIM(ISNULL(@p_Remark,''))),
                UpdateUser=@p_EmployeeID, UpdateDate=GETDATE()
            WHERE InspectionID=@p_InspectionID AND CompanyID=@p_CompanyID AND LocationID=@p_LocationID
              AND Status='DONE' AND IsApprove=0 AND ISNULL(ApprovedBy,'')='' AND ISNULL(ApprovedDate,CONVERT(DATETIME,'19000101',112))<=CONVERT(DATETIME,'19000101',112)
              AND ISNULL(ApprovalRemark,'')='' AND ISNULL(ReasonVoid,'')='';

            IF @@ROWCOUNT <> 1
                RAISERROR('Inspection approval state changed before this action completed. Refresh the queue.', 16, 1);
        END;

        COMMIT TRANSACTION;
        SELECT @p_InspectionID AS InspectionID, @p_Action AS Action,
               (SELECT FollowUpWONo FROM dbo.PIP_T_Inspection WHERE InspectionID=@p_InspectionID) AS FollowUpWONo;
    END TRY
    BEGIN CATCH
        IF XACT_STATE() <> 0 ROLLBACK TRANSACTION;
        DECLARE @V_ErrorMessage NVARCHAR(4000);
        DECLARE @V_ErrorSeverity INT;
        DECLARE @V_ErrorState INT;
        SELECT @V_ErrorMessage=ERROR_MESSAGE(), @V_ErrorSeverity=ERROR_SEVERITY(), @V_ErrorState=ERROR_STATE();
        RAISERROR(@V_ErrorMessage, @V_ErrorSeverity, @V_ErrorState);
    END CATCH;
END;
GO

CREATE PROCEDURE [dbo].[spUTL_PIP_T_Inspection_Batch_Sync]
    @p_ClientInspectionID VARCHAR(36),
    @p_CompanyID VARCHAR(10),
    @p_LocationID VARCHAR(3),
    @p_SectionID VARCHAR(10),
    @p_InspectionTypeCode VARCHAR(2),
    @p_InspectionTypeDetailCode VARCHAR(20),
    @p_ScheduledTime VARCHAR(5),
    @p_IDTemplate INT,
    @p_IDVersion INT,
    @p_TemplateNo VARCHAR(50),
    @p_TemplateVersion VARCHAR(10),
    @p_InspectionDate DATETIME,
    @p_StartDateTime DATETIME = NULL,
    @p_EndDateTime DATETIME = NULL,
    @p_Status VARCHAR(10),
    @p_Summary VARCHAR(500),
    @p_FollowUp VARCHAR(30),
    @p_FollowUpSection VARCHAR(10),
    @p_FollowUpRemark VARCHAR(300),
    @p_UpdateUser VARCHAR(50),
    @p_InspectionNo VARCHAR(10),
    @p_Points dbo.Type_N_Tbl_C13 READONLY,
    @p_Photos dbo.Type_N_Tbl_C5 READONLY,
    @p_Items dbo.Type_N_Tbl_C13 READONLY,
    @p_ItemValues dbo.Type_N_Tbl_C5 READONLY
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT ON;

    IF ISNULL(@p_Status,'') <> 'DONE'
    BEGIN
        RAISERROR('Only submitted BATCH inspections can be synchronized.',16,1);
        RETURN;
    END;

    IF NOT EXISTS
    (
        SELECT 1
        FROM dbo.PIP_M_InspectionType IT
        WHERE IT.InspectionTypeCode=@p_InspectionTypeCode
          AND IT.SectionID=@p_SectionID
          AND IT.IsActive=1
          AND ISNULL(IT.InspectionMode,'STANDARD')='BATCH'
    )
    BEGIN
        RAISERROR('Inspection Type is not configured for BATCH mode.',16,1);
        RETURN;
    END;

    IF NOT EXISTS (SELECT 1 FROM @p_Points)
    BEGIN
        RAISERROR('A BATCH inspection must contain at least one inspected point.',16,1);
        RETURN;
    END;

    IF EXISTS
    (
        SELECT C1 FROM @p_Points
        GROUP BY C1 HAVING COUNT(1)>1
    )
    BEGIN
        RAISERROR('Duplicate ClientPointID in BATCH inspection.',16,1);
        RETURN;
    END;

    IF EXISTS
    (
        SELECT 1
        FROM @p_Points P
        WHERE ISNULL(P.C1,'')=''
           OR ISNUMERIC(P.C2)=0
           OR NOT EXISTS
             (SELECT 1 FROM dbo.PIP_M_InspectionPoint M
              INNER JOIN dbo.PIP_M_InspectionMap MAP ON MAP.MapID=M.MapID
              WHERE M.InspectionPointID=CAST(P.C2 AS BIGINT)
                AND MAP.CompanyID=@p_CompanyID AND MAP.LocationID=@p_LocationID
                AND MAP.InspectionTypeCode=@p_InspectionTypeCode
                AND MAP.InspectionTypeDetailCode=@p_InspectionTypeDetailCode
                AND M.IsActive=1)
    )
    BEGIN
        RAISERROR('One or more Inspection Point records are invalid or inactive.',16,1);
        RETURN;
    END;

    IF EXISTS
    (
        SELECT 1
        FROM dbo.PIP_M_InspectionPoint M
        INNER JOIN dbo.PIP_M_InspectionMap MAP ON MAP.MapID=M.MapID
        WHERE MAP.CompanyID=@p_CompanyID AND MAP.LocationID=@p_LocationID
          AND MAP.InspectionTypeCode=@p_InspectionTypeCode
          AND MAP.InspectionTypeDetailCode=@p_InspectionTypeDetailCode
          AND M.IsActive=1
          AND NOT EXISTS
          (
              SELECT 1 FROM @p_Points P
              WHERE ISNUMERIC(P.C2)=1 AND CAST(P.C2 AS BIGINT)=M.InspectionPointID
          )
    )
    BEGIN
        RAISERROR('Every active Inspection Point for this BATCH must be submitted.',16,1);
        RETURN;
    END;

    IF EXISTS
    (
        SELECT 1
        FROM @p_Points P
        INNER JOIN dbo.PIP_M_InspectionPoint M ON M.InspectionPointID=CAST(P.C2 AS BIGINT)
        WHERE ISNULL(M.IsNormalAbnormalRequired,0)=1
          AND ISNULL(P.C6,'') NOT IN ('NORMAL','ABNORMAL')
    )
    BEGIN
        RAISERROR('Normal/Abnormal must be selected for every required Inspection Point.',16,1);
        RETURN;
    END;

    IF EXISTS
    (
        SELECT 1 FROM @p_Points
        WHERE UPPER(ISNULL(C6,''))='ABNORMAL'
          AND ISNULL(LTRIM(RTRIM(C5)),'')=''
    )
    BEGIN
        RAISERROR('Abnormal Remark is required when an Inspection Point is ABNORMAL.',16,1);
        RETURN;
    END;

    IF EXISTS
    (
        SELECT 1 FROM @p_Points
        WHERE UPPER(ISNULL(C6,''))='NORMAL'
          AND ISNULL(LTRIM(RTRIM(C5)),'')<>''
    )
    BEGIN
        RAISERROR('Abnormal Remark must be empty when an Inspection Point is NORMAL.',16,1);
        RETURN;
    END;

    IF EXISTS
    (
        SELECT 1 FROM @p_Points
        WHERE ISNULL(C4,'') NOT IN ('DONE','ABNORMAL')
    )
    BEGIN
        RAISERROR('All selected Inspection Points must be DONE or ABNORMAL before batch submission.',16,1);
        RETURN;
    END;

    IF EXISTS
    (
        SELECT 1
        FROM @p_Items V
        JOIN dbo.PIP_M_CheckListWOPM_Item I ON I.IDItem=CAST(V.C2 AS INT) AND I.ItemType='OPTION'
        WHERE ISNULL(I.IsValueRequired,0)=1 AND ISNULL(LTRIM(RTRIM(V.C5)),'')=''
    )
    BEGIN RAISERROR('A required OPTION item must be selected in every BATCH Inspection Point.',16,1); RETURN; END;

    IF EXISTS
    (
        SELECT 1
        FROM @p_Items V
        JOIN dbo.PIP_M_CheckListWOPM_Item I ON I.IDItem=CAST(V.C2 AS INT) AND I.ItemType='OPTION'
        WHERE ISNULL(LTRIM(RTRIM(V.C5)),'')<>''
          AND ISNULL(I.OptionList,'')<>''
          AND NOT EXISTS
          (
              SELECT 1 FROM dbo.SplitStringConvertToTable(I.OptionList,';') O
              WHERE UPPER(LTRIM(RTRIM(O.val)))=UPPER(LTRIM(RTRIM(V.C5)))
          )
    )
    BEGIN RAISERROR('A BATCH Inspection OPTION value is not in the configured option list.',16,1); RETURN; END;

    /* Validate native REPEAT values per submitted inspection point. */
    IF EXISTS
    (
        SELECT 1
        FROM @p_Items I
        WHERE UPPER(ISNULL(I.C4,''))='REPEAT'
          AND (NOT EXISTS (SELECT 1 FROM dbo.PIP_M_CheckListWOPM_ItemRepeat R WHERE R.IDItem=CAST(I.C2 AS INT))
               OR NOT EXISTS (SELECT 1 FROM dbo.PIP_M_CheckListWOPM_ItemRepeatField F WHERE F.IDItem=CAST(I.C2 AS INT)))
    )
    BEGIN RAISERROR('A BATCH REPEAT item has no master field configuration.',16,1); RETURN; END;

    IF EXISTS
    (
        SELECT 1 FROM @p_ItemValues V
        JOIN @p_Items I ON I.C1=V.C1 AND I.C2=V.C2 AND UPPER(ISNULL(I.C4,''))='REPEAT'
        WHERE NOT EXISTS
        (
            SELECT 1 FROM dbo.PIP_M_CheckListWOPM_ItemRepeatField F
            WHERE F.IDItem=CAST(V.C2 AS INT) AND F.FieldKey=V.C3
        )
    )
    BEGIN RAISERROR('Unknown REPEAT child FieldKey submitted for BATCH inspection.',16,1); RETURN; END;

    IF EXISTS
    (
        SELECT 1
        FROM @p_Items I
        JOIN dbo.PIP_M_CheckListWOPM_ItemRepeat R ON R.IDItem=CAST(I.C2 AS INT)
        WHERE UPPER(ISNULL(I.C4,''))='REPEAT'
          AND
          (
              (R.RepeatMode='FIELDS' AND
                ((SELECT COUNT(DISTINCT CAST(V.C5 AS INT)) FROM @p_ItemValues V WHERE V.C1=I.C1 AND V.C2=I.C2)<>1
                 OR EXISTS (SELECT 1 FROM @p_ItemValues V WHERE V.C1=I.C1 AND V.C2=I.C2 AND CAST(V.C5 AS INT)<>0)))
              OR (R.RepeatMode='OBJECTS' AND R.CountMode='FIXED' AND
                (SELECT COUNT(DISTINCT CAST(V.C5 AS INT)) FROM @p_ItemValues V WHERE V.C1=I.C1 AND V.C2=I.C2)<>R.FixedCount)
              OR (R.RepeatMode='OBJECTS' AND R.CountMode='FLEX' AND R.MinCount IS NOT NULL AND
                (SELECT COUNT(DISTINCT CAST(V.C5 AS INT)) FROM @p_ItemValues V WHERE V.C1=I.C1 AND V.C2=I.C2)<R.MinCount)
              OR (R.RepeatMode='OBJECTS' AND R.CountMode='FLEX' AND R.MaxCount IS NOT NULL AND
                (SELECT COUNT(DISTINCT CAST(V.C5 AS INT)) FROM @p_ItemValues V WHERE V.C1=I.C1 AND V.C2=I.C2)>R.MaxCount)
          )
    )
    BEGIN RAISERROR('REPEAT object count does not match BATCH template configuration.',16,1); RETURN; END;

    IF EXISTS
    (
        SELECT 1
        FROM @p_ItemValues V
        JOIN @p_Items I ON I.C1=V.C1 AND I.C2=V.C2 AND UPPER(ISNULL(I.C4,''))='REPEAT'
        WHERE CAST(ISNULL(NULLIF(V.C5,''),'0') AS INT)<0
    )
    BEGIN RAISERROR('BATCH REPEAT ObjectIndex cannot be negative.',16,1); RETURN; END;

    IF EXISTS
    (
        SELECT V.C1,V.C2
        FROM @p_ItemValues V
        JOIN @p_Items I ON I.C1=V.C1 AND I.C2=V.C2 AND UPPER(ISNULL(I.C4,''))='REPEAT'
        JOIN dbo.PIP_M_CheckListWOPM_ItemRepeat R ON R.IDItem=CAST(I.C2 AS INT) AND R.RepeatMode='OBJECTS'
        GROUP BY V.C1,V.C2
        HAVING MIN(CAST(ISNULL(NULLIF(V.C5,''),'0') AS INT))<>0
            OR MAX(CAST(ISNULL(NULLIF(V.C5,''),'0') AS INT))<>COUNT(DISTINCT CAST(ISNULL(NULLIF(V.C5,''),'0') AS INT))-1
    )
    BEGIN RAISERROR('BATCH REPEAT ObjectIndex values must be sequential from zero.',16,1); RETURN; END;

    IF EXISTS
    (
        SELECT 1
        FROM @p_Items I
        JOIN dbo.PIP_M_CheckListWOPM_ItemRepeatField F ON F.IDItem=CAST(I.C2 AS INT) AND F.IsRequired=1
        JOIN (SELECT DISTINCT C1,C2,C5 FROM @p_ItemValues) O ON O.C1=I.C1 AND O.C2=I.C2
        LEFT JOIN @p_ItemValues V ON V.C1=O.C1 AND V.C2=O.C2 AND V.C5=O.C5 AND V.C3=F.FieldKey
        WHERE UPPER(ISNULL(I.C4,''))='REPEAT' AND ISNULL(LTRIM(RTRIM(V.C4)),'')=''
    )
    BEGIN RAISERROR('A required REPEAT child field is missing in BATCH inspection.',16,1); RETURN; END;

    IF EXISTS
    (
        SELECT 1
        FROM @p_ItemValues V
        JOIN dbo.PIP_M_CheckListWOPM_ItemRepeatField F
          ON F.IDItem=CAST(V.C2 AS INT) AND F.FieldKey=V.C3 AND F.FieldType='OPTION'
        WHERE ISNULL(LTRIM(RTRIM(V.C4)),'')<>''
          AND NOT EXISTS
          (
              SELECT 1 FROM dbo.SplitStringConvertToTable(F.OptionList,';') O
              WHERE UPPER(LTRIM(RTRIM(O.val)))=UPPER(LTRIM(RTRIM(V.C4)))
          )
    )
    BEGIN RAISERROR('A BATCH REPEAT OPTION value is not in the configured option list.',16,1); RETURN; END;

    IF EXISTS
    (
        SELECT 1 FROM @p_ItemValues V
        JOIN dbo.PIP_M_CheckListWOPM_ItemRepeatField F ON F.IDItem=CAST(V.C2 AS INT) AND F.FieldKey=V.C3
        WHERE F.FieldType IN ('NUMBER','RANGE') AND ISNULL(LTRIM(RTRIM(V.C4)),'')<>'' AND ISNUMERIC(V.C4)=0
    )
    BEGIN RAISERROR('A BATCH REPEAT numeric value must be a valid number.',16,1); RETURN; END;

    IF EXISTS
    (
        SELECT 1 FROM @p_ItemValues V
        JOIN dbo.PIP_M_CheckListWOPM_ItemRepeatField F ON F.IDItem=CAST(V.C2 AS INT) AND F.FieldKey=V.C3
        WHERE F.MinValue IS NOT NULL AND ISNULL(LTRIM(RTRIM(V.C4)),'')<>''
          AND (CONVERT(FLOAT,V.C4)<F.MinValue OR (F.MinInclusive=0 AND CONVERT(FLOAT,V.C4)=F.MinValue))
    )
    BEGIN RAISERROR('A BATCH REPEAT numeric value is below its minimum.',16,1); RETURN; END;

    IF EXISTS
    (
        SELECT 1 FROM @p_ItemValues V
        JOIN dbo.PIP_M_CheckListWOPM_ItemRepeatField F ON F.IDItem=CAST(V.C2 AS INT) AND F.FieldKey=V.C3
        WHERE F.MaxValue IS NOT NULL AND ISNULL(LTRIM(RTRIM(V.C4)),'')<>''
          AND (CONVERT(FLOAT,V.C4)>F.MaxValue OR (F.MaxInclusive=0 AND CONVERT(FLOAT,V.C4)=F.MaxValue))
    )
    BEGIN RAISERROR('A BATCH REPEAT numeric value exceeds its maximum.',16,1); RETURN; END;

    DECLARE @V_EntryTranCount INT;
    SET @V_EntryTranCount=@@TRANCOUNT;

    BEGIN TRY
        IF @V_EntryTranCount=0 BEGIN TRANSACTION;
        ELSE SAVE TRANSACTION InspectionBatchSyncSave;

        DECLARE @V_InspectionNo VARCHAR(10);
        SET @V_InspectionNo=NULLIF(LTRIM(RTRIM(ISNULL(@p_InspectionNo,''))), '');

        /* The canonical Inspection sync generates InspectionNo when this parameter is empty.
           Batch inspections use the same InspectionNo mechanism; no separate batch sequence is required. */

        /* Reuse the current canonical header sync procedure. Empty checklist/photo
           TVPs ensure no batch-level results are written; all child data is saved below. */
        DECLARE @V_EmptyPhotos dbo.Type_N_Tbl_C5;
        DECLARE @V_EmptyItems dbo.Type_N_Tbl_C13;
        DECLARE @V_EmptyValues dbo.Type_N_Tbl_C5;

        EXEC dbo.spUTL_PIP_T_Inspection_Sync
            @p_ClientInspectionID=@p_ClientInspectionID,
            @p_CompanyID=@p_CompanyID,
            @p_LocationID=@p_LocationID,
            @p_SectionID=@p_SectionID,
            @p_InspectionTypeCode=@p_InspectionTypeCode,
            @p_InspectionTypeDetailCode=@p_InspectionTypeDetailCode,
            @p_ScheduledTime=@p_ScheduledTime,
            @p_IDTemplate=@p_IDTemplate,
            @p_IDVersion=@p_IDVersion,
            @p_TemplateNo=@p_TemplateNo,
            @p_TemplateVersion=@p_TemplateVersion,
            @p_InspectionDate=@p_InspectionDate,
            @p_StartDateTime=@p_StartDateTime,
            @p_EndDateTime=@p_EndDateTime,
            @p_Status=@p_Status,
            @p_Summary=@p_Summary,
            @p_FollowUp=@p_FollowUp,
            @p_FollowUpSection=@p_FollowUpSection,
            @p_FollowUpRemark=@p_FollowUpRemark,
            @p_UpdateUser=@p_UpdateUser,
            @p_InspectionNo=@V_InspectionNo,
            @p_Photos=@V_EmptyPhotos,
            @p_Items=@V_EmptyItems,
            @p_ItemValues=@V_EmptyValues;

        DECLARE @V_InspectionID BIGINT;
        SELECT @V_InspectionID=InspectionID
        FROM dbo.PIP_T_Inspection WITH (UPDLOCK,HOLDLOCK)
        WHERE ClientInspectionID=@p_ClientInspectionID;

        IF @V_InspectionID IS NULL
            RAISERROR('BATCH header was not saved by the canonical Inspection sync procedure.',16,1);

        /* Remove old child results/photos before rebuilding the submitted local snapshot. */
        DELETE V
        FROM dbo.PIP_T_InspectionResultValue V
        JOIN dbo.PIP_T_InspectionResult R ON R.InspectionResultID=V.InspectionResultID
        WHERE R.InspectionID=@V_InspectionID;

        DELETE FROM dbo.PIP_T_InspectionResult WHERE InspectionID=@V_InspectionID;
        DELETE FROM dbo.PIP_T_InspectionPhoto WHERE InspectionID=@V_InspectionID;

        DECLARE @V_PointMap TABLE
        (
            ClientPointID VARCHAR(36) NOT NULL PRIMARY KEY,
            InspectionPointTransactionID BIGINT NOT NULL
        );

        UPDATE T
        SET T.InspectionPointID=CAST(P.C2 AS BIGINT),
            T.InspectionDateTime=ISNULL(CONVERT(DATETIME,NULLIF(P.C3,''),126),GETDATE()),
            T.Status=P.C4,
            T.AbnormalRemark=NULLIF(LTRIM(RTRIM(ISNULL(P.C5,''))),''),
            T.UpdateUser=@p_UpdateUser,
            T.UpdateDate=GETDATE()
        FROM dbo.PIP_T_InspectionPoint T
        JOIN @p_Points P ON P.C1=T.ClientPointID
        WHERE T.InspectionID=@V_InspectionID;

        DECLARE @V_NextPointID BIGINT;
        SELECT @V_NextPointID=ISNULL(MAX(InspectionPointTransactionID),0)
        FROM dbo.PIP_T_InspectionPoint WITH (UPDLOCK,HOLDLOCK);

        ;WITH NewPoints AS
        (
            SELECT P.C1 ClientPointID, CAST(P.C2 AS BIGINT) InspectionPointID,
                   ISNULL(CONVERT(DATETIME,NULLIF(P.C3,''),126),GETDATE()) InspectionDateTime,
                   P.C4 Status,
                   NULLIF(LTRIM(RTRIM(ISNULL(P.C5,''))),'') AbnormalRemark,
                   @V_NextPointID+ROW_NUMBER() OVER (ORDER BY P.C1) NewID
            FROM @p_Points P
            WHERE NOT EXISTS
            (
                SELECT 1 FROM dbo.PIP_T_InspectionPoint T
                WHERE T.InspectionID=@V_InspectionID AND T.ClientPointID=P.C1
            )
        )
        INSERT dbo.PIP_T_InspectionPoint
        (
            InspectionPointTransactionID,InspectionID,ClientPointID,InspectionPointID,
            InspectionDateTime,Status,AbnormalRemark,EntryUser,EntryDate,UpdateUser,UpdateDate
        )
        OUTPUT inserted.ClientPointID, inserted.InspectionPointTransactionID
            INTO @V_PointMap(ClientPointID,InspectionPointTransactionID)
        SELECT N.NewID,@V_InspectionID,N.ClientPointID,N.InspectionPointID,
               N.InspectionDateTime,N.Status,N.AbnormalRemark,@p_UpdateUser,GETDATE(),@p_UpdateUser,GETDATE()
        FROM NewPoints N;

        INSERT @V_PointMap(ClientPointID,InspectionPointTransactionID)
        SELECT T.ClientPointID,T.InspectionPointTransactionID
        FROM dbo.PIP_T_InspectionPoint T
        JOIN @p_Points P ON P.C1=T.ClientPointID
        WHERE T.InspectionID=@V_InspectionID
          AND NOT EXISTS (SELECT 1 FROM @V_PointMap M WHERE M.ClientPointID=T.ClientPointID);

        DELETE T
        FROM dbo.PIP_T_InspectionPoint T
        WHERE T.InspectionID=@V_InspectionID
          AND NOT EXISTS (SELECT 1 FROM @p_Points P WHERE P.C1=T.ClientPointID);

        DECLARE @V_ResultMap TABLE
        (
            InspectionPointTransactionID BIGINT NOT NULL,
            IDItem INT NOT NULL,
            InspectionResultID BIGINT NOT NULL,
            PRIMARY KEY(InspectionPointTransactionID,IDItem)
        );
        DECLARE @V_NextResultID BIGINT;
        SELECT @V_NextResultID=ISNULL(MAX(InspectionResultID),0)
        FROM dbo.PIP_T_InspectionResult WITH (UPDLOCK,HOLDLOCK);

        ;WITH ResultRows AS
        (
            SELECT I.C1 ClientPointID, CAST(I.C2 AS INT) IDItem,
                   I.C3 ItemLabel, I.C4 ItemType, ISNULL(I.C5,'') Value,
                   ISNULL(I.C6,'') Remark1,
                   CASE WHEN ISNULL(I.C7,'0') IN ('1','True','true') THEN 1 ELSE 0 END IsAnomali,
                   ISNULL(CAST(NULLIF(I.C8,'') AS INT),0) DisplayOrder,
                   ISNULL(I.C9,'') Remark2,
                   @V_NextResultID+ROW_NUMBER() OVER (ORDER BY I.C1,CAST(I.C2 AS INT)) NewID
            FROM @p_Items I
        )
        INSERT dbo.PIP_T_InspectionResult
        (
            InspectionResultID,InspectionID,InspectionPointTransactionID,IDItem,ItemLabel,ItemType,
            Value,Remark1,Remark2,IsAnomali,DisplayOrder,EntryUser,EntryDate,UpdateUser,UpdateDate
        )
        OUTPUT inserted.InspectionPointTransactionID, inserted.IDItem, inserted.InspectionResultID
            INTO @V_ResultMap(InspectionPointTransactionID,IDItem,InspectionResultID)
        SELECT R.NewID,@V_InspectionID,PM.InspectionPointTransactionID,R.IDItem,R.ItemLabel,R.ItemType,
               R.Value,R.Remark1,R.Remark2,R.IsAnomali,R.DisplayOrder,
               @p_UpdateUser,GETDATE(),@p_UpdateUser,GETDATE()
        FROM ResultRows R
        JOIN @V_PointMap PM ON PM.ClientPointID=R.ClientPointID;

        INSERT dbo.PIP_T_InspectionResultValue
        (
            InspectionResultID,ObjectIndex,FieldName,Value,EntryUser,EntryDate,UpdateUser,UpdateDate
        )
        SELECT RM.InspectionResultID,CAST(ISNULL(NULLIF(V.C5,''),'0') AS INT),V.C3,ISNULL(V.C4,''),
               @p_UpdateUser,GETDATE(),@p_UpdateUser,GETDATE()
        FROM @p_ItemValues V
        JOIN @V_PointMap PM ON PM.ClientPointID=V.C1
        JOIN @V_ResultMap RM ON RM.InspectionPointTransactionID=PM.InspectionPointTransactionID
                            AND RM.IDItem=CAST(V.C2 AS INT);

        DECLARE @V_NextPhotoID BIGINT;
        SELECT @V_NextPhotoID=ISNULL(MAX(InspectionPhotoID),0)
        FROM dbo.PIP_T_InspectionPhoto WITH (UPDLOCK,HOLDLOCK);

        ;WITH PhotoRows AS
        (
            SELECT P.C1 ClientPointID,P.C2 PhotoType,P.C3 FileName,P.C4 FileType,
                   CAST(ISNULL(NULLIF(P.C5,''),'0') AS INT) SeqNo,
                   @V_NextPhotoID+ROW_NUMBER() OVER (ORDER BY P.C1,CAST(ISNULL(NULLIF(P.C5,''),'0') AS INT),P.C3) NewID
            FROM @p_Photos P
        )
        INSERT dbo.PIP_T_InspectionPhoto
        (
            InspectionPhotoID,InspectionID,InspectionPointTransactionID,PhotoType,
            FileName,FileType,SeqNo,EntryUser,EntryDate
        )
        SELECT P.NewID,@V_InspectionID,PM.InspectionPointTransactionID,
               P.PhotoType,P.FileName,P.FileType,P.SeqNo,@p_UpdateUser,GETDATE()
        FROM PhotoRows P
        JOIN @V_PointMap PM ON PM.ClientPointID=P.ClientPointID;

        IF @V_EntryTranCount=0 COMMIT TRANSACTION;

        SELECT @V_InspectionID InspectionID,@p_ClientInspectionID ClientInspectionID,
               InspectionNo
        FROM dbo.PIP_T_Inspection
        WHERE InspectionID=@V_InspectionID;
    END TRY
    BEGIN CATCH
        IF XACT_STATE()=1
        BEGIN
            IF @V_EntryTranCount=0 ROLLBACK TRANSACTION;
            ELSE ROLLBACK TRANSACTION InspectionBatchSyncSave;
        END
        ELSE IF XACT_STATE()=-1 AND @V_EntryTranCount=0
            ROLLBACK TRANSACTION;

        DECLARE @V_ErrorMessage NVARCHAR(4000);
        DECLARE @V_ErrorSeverity INT;
        DECLARE @V_ErrorState INT;
        SELECT @V_ErrorMessage=ERROR_MESSAGE(),@V_ErrorSeverity=ERROR_SEVERITY(),@V_ErrorState=ERROR_STATE();
        RAISERROR(@V_ErrorMessage,@V_ErrorSeverity,@V_ErrorState);
    END CATCH;
END;
GO

CREATE PROCEDURE [dbo].[spUTL_PIP_T_Inspection_Sync]
(
    @p_ClientInspectionID VARCHAR(36),
    @p_CompanyID VARCHAR(10),
    @p_LocationID VARCHAR(3),
    @p_SectionID VARCHAR(10),
    @p_InspectionTypeCode VARCHAR(2),
    @p_InspectionTypeDetailCode VARCHAR(20),
    @p_ScheduledTime VARCHAR(5),
    @p_IDTemplate INT,
    @p_IDVersion INT,
    @p_TemplateNo VARCHAR(50),
    @p_TemplateVersion VARCHAR(10),
    @p_InspectionDate DATETIME,
    @p_StartDateTime DATETIME = NULL,
    @p_EndDateTime DATETIME = NULL,
    @p_Status VARCHAR(10),
    @p_Summary VARCHAR(500),
    @p_FollowUp VARCHAR(30),
    @p_FollowUpSection VARCHAR(10),
    @p_FollowUpRemark VARCHAR(300),
    @p_UpdateUser VARCHAR(50),
    @p_InspectionNo VARCHAR(10),
    @p_Photos dbo.Type_N_Tbl_C5 READONLY,
    @p_Items dbo.Type_N_Tbl_C13 READONLY,
    @p_ItemValues dbo.Type_N_Tbl_C5 READONLY
)
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT ON;

    SET @p_ScheduledTime = ISNULL(@p_ScheduledTime, '');
    SET @p_StartDateTime = ISNULL(@p_StartDateTime, CONVERT(DATETIME,'19000101',112));
    SET @p_EndDateTime = ISNULL(@p_EndDateTime, CONVERT(DATETIME,'19000101',112));
    SET @p_Summary = ISNULL(@p_Summary, '');
    SET @p_FollowUp = ISNULL(NULLIF(LTRIM(RTRIM(@p_FollowUp)), ''), 'No Follow Up');
    SET @p_FollowUpSection = ISNULL(@p_FollowUpSection, '');
    SET @p_FollowUpRemark = ISNULL(@p_FollowUpRemark, '');

    /* Map shared generic TVPs to the inspection-specific row shape. */
    DECLARE @tbl_Items TABLE
    (
        IDItem INT NOT NULL,
        ItemLabel VARCHAR(500) NOT NULL,
        ItemType VARCHAR(20) NOT NULL,
        Value VARCHAR(50) NULL,
        Remark1 VARCHAR(500) NULL,
        Remark2 VARCHAR(500) NULL,
        IsAnomali BIT NOT NULL,
        DisplayOrder INT NOT NULL
    );
    INSERT INTO @tbl_Items (IDItem, ItemLabel, ItemType, Value, Remark1, IsAnomali, DisplayOrder, Remark2)
    SELECT CAST(C1 AS INT), ISNULL(C2,''), ISNULL(C3,''), ISNULL(C4,''), ISNULL(C5,''),
           CAST(ISNULL(NULLIF(C6, ''), '0') AS BIT),
           CAST(ISNULL(NULLIF(C7, ''), '0') AS INT), ISNULL(C8,'')
    FROM @p_Items;

    DECLARE @tbl_ItemValues TABLE
    (
        IDItem INT NOT NULL,
        ObjectIndex INT NOT NULL,
        FieldName VARCHAR(100) NOT NULL,
        Value NVARCHAR(500) NULL
    );
    INSERT INTO @tbl_ItemValues (IDItem, ObjectIndex, FieldName, Value)
    SELECT CAST(C1 AS INT), CAST(ISNULL(NULLIF(C4,''),'0') AS INT), ISNULL(C2,''), ISNULL(C3,'')
    FROM @p_ItemValues;

    /* Server-side REPEAT validation for STANDARD sync. BATCH validates each point in its
       canonical C# validation path and persists the same ObjectIndex shape below. */
    IF EXISTS
    (
        SELECT 1 FROM @tbl_Items I
        WHERE I.ItemType='REPEAT'
          AND (NOT EXISTS (SELECT 1 FROM dbo.PIP_M_CheckListWOPM_ItemRepeat R WHERE R.IDItem=I.IDItem)
               OR NOT EXISTS (SELECT 1 FROM dbo.PIP_M_CheckListWOPM_ItemRepeatField F WHERE F.IDItem=I.IDItem))
    )
    BEGIN RAISERROR('REPEAT checklist item has no valid master configuration.',16,1); RETURN; END;

    IF EXISTS
    (
        SELECT 1
        FROM @tbl_ItemValues V
        JOIN @tbl_Items I ON I.IDItem=V.IDItem AND I.ItemType='REPEAT'
        WHERE NOT EXISTS
        (
            SELECT 1 FROM dbo.PIP_M_CheckListWOPM_ItemRepeatField F
            WHERE F.IDItem=V.IDItem AND F.FieldKey=V.FieldName
        )
    )
    BEGIN RAISERROR('Unknown REPEAT child FieldKey submitted.',16,1); RETURN; END;

    IF EXISTS
    (
        SELECT 1
        FROM @tbl_Items I
        JOIN dbo.PIP_M_CheckListWOPM_ItemRepeat R ON R.IDItem=I.IDItem
        WHERE I.ItemType='REPEAT'
          AND R.RepeatMode='FIELDS'
          AND
          (
              (SELECT COUNT(DISTINCT V.ObjectIndex) FROM @tbl_ItemValues V WHERE V.IDItem=I.IDItem)<>1
              OR EXISTS (SELECT 1 FROM @tbl_ItemValues V WHERE V.IDItem=I.IDItem AND V.ObjectIndex<>0)
          )
    )
    BEGIN RAISERROR('REPEAT FIELDS must contain exactly ObjectIndex 0.',16,1); RETURN; END;

    IF EXISTS
    (
        SELECT 1
        FROM @tbl_Items I
        JOIN dbo.PIP_M_CheckListWOPM_ItemRepeat R ON R.IDItem=I.IDItem
        WHERE I.ItemType='REPEAT' AND R.RepeatMode='OBJECTS'
          AND
          (
              (R.CountMode='FIXED' AND (SELECT COUNT(DISTINCT V.ObjectIndex) FROM @tbl_ItemValues V WHERE V.IDItem=I.IDItem)<>R.FixedCount)
              OR (R.CountMode='FLEX' AND R.MinCount IS NOT NULL AND (SELECT COUNT(DISTINCT V.ObjectIndex) FROM @tbl_ItemValues V WHERE V.IDItem=I.IDItem)<R.MinCount)
              OR (R.CountMode='FLEX' AND R.MaxCount IS NOT NULL AND (SELECT COUNT(DISTINCT V.ObjectIndex) FROM @tbl_ItemValues V WHERE V.IDItem=I.IDItem)>R.MaxCount)
          )
    )
    BEGIN RAISERROR('REPEAT OBJECTS count does not match its master configuration.',16,1); RETURN; END;

    IF EXISTS
    (
        SELECT 1
        FROM @tbl_ItemValues V
        JOIN @tbl_Items I ON I.IDItem=V.IDItem AND I.ItemType='REPEAT'
        WHERE V.ObjectIndex<0
    )
    BEGIN RAISERROR('REPEAT ObjectIndex cannot be negative.',16,1); RETURN; END;

    IF EXISTS
    (
        SELECT 1
        FROM @tbl_Items I
        JOIN dbo.PIP_M_CheckListWOPM_ItemRepeat R ON R.IDItem=I.IDItem
        JOIN @tbl_ItemValues V ON V.IDItem=I.IDItem
        WHERE I.ItemType='REPEAT' AND R.RepeatMode='OBJECTS'
        GROUP BY I.IDItem
        HAVING MIN(V.ObjectIndex)<>0 OR MAX(V.ObjectIndex)<>COUNT(DISTINCT V.ObjectIndex)-1
    )
    BEGIN RAISERROR('REPEAT OBJECTS ObjectIndex values must be sequential from zero.',16,1); RETURN; END;

    IF EXISTS
    (
        SELECT 1
        FROM @tbl_Items I
        JOIN dbo.PIP_M_CheckListWOPM_ItemRepeatField F ON F.IDItem=I.IDItem AND F.IsRequired=1
        JOIN (SELECT DISTINCT IDItem,ObjectIndex FROM @tbl_ItemValues) O ON O.IDItem=I.IDItem
        LEFT JOIN @tbl_ItemValues V ON V.IDItem=O.IDItem AND V.ObjectIndex=O.ObjectIndex AND V.FieldName=F.FieldKey
        WHERE I.ItemType='REPEAT' AND ISNULL(LTRIM(RTRIM(V.Value)),'')=''
    )
    BEGIN RAISERROR('A required REPEAT child field is missing.',16,1); RETURN; END;

    IF EXISTS
    (
        SELECT 1
        FROM @tbl_ItemValues V
        JOIN dbo.PIP_M_CheckListWOPM_ItemRepeatField F
          ON F.IDItem=V.IDItem AND F.FieldKey=V.FieldName AND F.FieldType='OPTION'
        WHERE ISNULL(LTRIM(RTRIM(V.Value)),'')<>''
          AND NOT EXISTS
          (
              SELECT 1 FROM dbo.SplitStringConvertToTable(F.OptionList,';') O
              WHERE UPPER(LTRIM(RTRIM(O.val)))=UPPER(LTRIM(RTRIM(V.Value)))
          )
    )
    BEGIN RAISERROR('A REPEAT OPTION value is not in the configured option list.',16,1); RETURN; END;

    IF EXISTS
    (
        SELECT 1
        FROM @p_Items V
        JOIN dbo.PIP_M_CheckListWOPM_Item I ON I.IDItem=CAST(V.C1 AS INT)
        WHERE I.ItemType='OPTION' AND ISNULL(I.IsValueRequired,0)=1
          AND ISNULL(LTRIM(RTRIM(V.C4)),'')=''
    )
    BEGIN RAISERROR('A required OPTION checklist item must be selected before Inspection sync.',16,1); RETURN; END;

    IF EXISTS
    (
        SELECT 1
        FROM @p_Items V
        JOIN dbo.PIP_M_CheckListWOPM_Item I ON I.IDItem=CAST(V.C1 AS INT) AND I.ItemType='OPTION'
        WHERE ISNULL(LTRIM(RTRIM(V.C4)),'')<>''
          AND ISNULL(I.OptionList,'')<>''
          AND NOT EXISTS
          (
              SELECT 1 FROM dbo.SplitStringConvertToTable(I.OptionList,';') O
              WHERE UPPER(LTRIM(RTRIM(O.val)))=UPPER(LTRIM(RTRIM(V.C4)))
          )
    )
    BEGIN RAISERROR('An Inspection OPTION value is not in the configured option list.',16,1); RETURN; END;

    IF EXISTS
    (
        SELECT 1 FROM @tbl_ItemValues V
        JOIN dbo.PIP_M_CheckListWOPM_ItemRepeatField F ON F.IDItem=V.IDItem AND F.FieldKey=V.FieldName
        WHERE F.FieldType IN ('NUMBER','RANGE') AND ISNULL(LTRIM(RTRIM(V.Value)),'')<>'' AND ISNUMERIC(V.Value)=0
    )
    BEGIN RAISERROR('A REPEAT numeric value must be a valid number.',16,1); RETURN; END;

    IF EXISTS
    (
        SELECT 1 FROM @tbl_ItemValues V
        JOIN dbo.PIP_M_CheckListWOPM_ItemRepeatField F ON F.IDItem=V.IDItem AND F.FieldKey=V.FieldName
        WHERE F.MinValue IS NOT NULL AND ISNULL(LTRIM(RTRIM(V.Value)),'')<>''
          AND (CONVERT(FLOAT,V.Value)<F.MinValue OR (F.MinInclusive=0 AND CONVERT(FLOAT,V.Value)=F.MinValue))
    )
    BEGIN RAISERROR('A REPEAT numeric value is below its minimum.',16,1); RETURN; END;

    IF EXISTS
    (
        SELECT 1 FROM @tbl_ItemValues V
        JOIN dbo.PIP_M_CheckListWOPM_ItemRepeatField F ON F.IDItem=V.IDItem AND F.FieldKey=V.FieldName
        WHERE F.MaxValue IS NOT NULL AND ISNULL(LTRIM(RTRIM(V.Value)),'')<>''
          AND (CONVERT(FLOAT,V.Value)>F.MaxValue OR (F.MaxInclusive=0 AND CONVERT(FLOAT,V.Value)=F.MaxValue))
    )
    BEGIN RAISERROR('A REPEAT numeric value exceeds its maximum.',16,1); RETURN; END;

    /* Drafts are device-local only. SQL may persist completed inspections only. */
    IF ISNULL(@p_Status, '') <> 'DONE'
    BEGIN
        RAISERROR('Only completed inspections (DONE) can be saved to SQL Server.', 16, 1);
        RETURN;
    END;

    SET @p_FollowUp = LTRIM(RTRIM(ISNULL(@p_FollowUp, 'No Follow Up')));
    SET @p_FollowUpSection = LTRIM(RTRIM(ISNULL(@p_FollowUpSection, '')));
    SET @p_FollowUpRemark = LTRIM(RTRIM(ISNULL(@p_FollowUpRemark, '')));

    /* Follow Up is controlled by the selected checklist Version for both STANDARD and BATCH.
       BATCH delegates its header to this canonical procedure, so one rule covers both modes. */
    DECLARE @V_IsFollowUpRequired BIT;
    SELECT @V_IsFollowUpRequired=ISNULL(V.IsFollowUpRequired,1)
    FROM dbo.PIP_M_CheckListWOPM_Version V
    WHERE V.IDVersion=@p_IDVersion AND V.IDTemplate=@p_IDTemplate;
    IF ISNULL(@V_IsFollowUpRequired,1)=0
    BEGIN
        SET @p_FollowUp='No Follow Up';
        SET @p_FollowUpSection='';
        SET @p_FollowUpRemark='';
    END;

    IF @p_FollowUp NOT IN ('No Follow Up', 'Recommend Follow Up')
    BEGIN
        RAISERROR('Invalid Inspection Follow Up selection.', 16, 1);
        RETURN;
    END;

    IF @p_FollowUp = 'Recommend Follow Up'
    BEGIN
        IF @p_Status = 'DONE'
        BEGIN
            IF @p_FollowUpSection NOT IN ('ACON','COMM','MECH','ELEC','CIVIL','PFIT','PHKP','PSEC')
            BEGIN
                RAISERROR('A valid Responsible Section is required for Follow Up.', 16, 1);
                RETURN;
            END;
            IF @p_FollowUpRemark = ''
            BEGIN
                RAISERROR('Follow Up Work Order Description is required.', 16, 1);
                RETURN;
            END;
        END;
    END
    ELSE
    BEGIN
        SET @p_FollowUpSection = '';
        SET @p_FollowUpRemark = '';
    END;

    IF NOT EXISTS
    (
        SELECT 1
        FROM dbo.UTL_User U
        WHERE U.EmployeeID=@p_UpdateUser
          AND
          (
              EXISTS
              (
                  SELECT 1 FROM dbo.SysWebUserCompanyLocation C
                  WHERE C.UserID=U.UserID AND C.AppID='pip'
                    AND C.CompanyID=@p_CompanyID AND C.LocationID=@p_LocationID
              )
              OR EXISTS
              (
                  SELECT 1 FROM dbo.F_SelectOneLogin('pip', U.EmployeeID, '') L
                  WHERE L.CompanyID=@p_CompanyID AND L.LocationID=@p_LocationID
              )
          )
    )
    BEGIN
        RAISERROR('Employee is not authorized for this company/location.', 16, 1);
        RETURN;
    END;

    IF NOT EXISTS
    (
        SELECT 1
        FROM dbo.UTL_User U
        WHERE U.EmployeeID=@p_UpdateUser
          AND ISNULL(U.IsWO,0)=1
          AND EXISTS
              (SELECT 1 FROM dbo.SplitStringConvertToTable(U.Section, ';') S
               WHERE S.val=@p_SectionID)
    )
    BEGIN
        RAISERROR('Employee is not authorized for this inspection section.', 16, 1);
        RETURN;
    END;

    IF NOT EXISTS
    (
        SELECT 1
        FROM dbo.PIP_M_InspectionType T
        JOIN dbo.PIP_M_InspectionTypeDetail D
          ON D.InspectionTypeCode=T.InspectionTypeCode
        WHERE T.InspectionTypeCode=@p_InspectionTypeCode
          AND D.InspectionTypeDetailCode=@p_InspectionTypeDetailCode
          AND D.CompanyID=@p_CompanyID
          AND D.LocationID=@p_LocationID
          AND T.SectionID=@p_SectionID
          AND T.IsActive=1 AND D.IsActive=1
    )
    BEGIN
        RAISERROR('Inspection Type or detail is inactive, missing, or belongs to another section.', 16, 1);
        RETURN;
    END;

    IF EXISTS
    (
        SELECT 1
        FROM dbo.PIP_M_InspectionTypeDetail D
        WHERE D.InspectionTypeCode=@p_InspectionTypeCode
          AND D.InspectionTypeDetailCode=@p_InspectionTypeDetailCode
          AND D.CompanyID=@p_CompanyID
          AND D.LocationID=@p_LocationID
          AND LEN(LTRIM(RTRIM(ISNULL(D.ScheduleTime,'')))) > 0
    )
       AND ISNULL(@p_ScheduledTime, '') = ''
    BEGIN
        RAISERROR('Scheduled time is required for this inspection detail.', 16, 1);
        RETURN;
    END;

    IF ISNULL(@p_ScheduledTime, '') <> ''
       AND NOT EXISTS
       (
           SELECT 1
           FROM dbo.PIP_M_InspectionTypeDetail D
           WHERE D.InspectionTypeCode=@p_InspectionTypeCode
             AND D.InspectionTypeDetailCode=@p_InspectionTypeDetailCode
             AND D.CompanyID=@p_CompanyID
             AND D.LocationID=@p_LocationID
             AND CHARINDEX(';' + @p_ScheduledTime + ';', ';' + REPLACE(D.ScheduleTime, ' ', '') + ';') > 0
       )
    BEGIN
        RAISERROR('Scheduled time is not configured for this inspection detail.', 16, 1);
        RETURN;
    END;

    /* A returned, unapproved inspection may change parameters/template.
       The same ClientInspectionID is updated and its checklist/photo snapshot is replaced atomically. */

    IF NOT EXISTS
    (
        SELECT 1
        FROM dbo.PIP_M_CheckListWOPM T
        JOIN dbo.PIP_M_CheckListWOPM_Version V ON V.IDVersion=@p_IDVersion AND V.IDTemplate=T.IDTemplate
        WHERE T.IDTemplate=@p_IDTemplate
          AND T.TemplateNo=@p_TemplateNo
          AND T.SectionID=@p_SectionID
          AND T.TemplateType='I'
          AND T.InspectionTypeCode=@p_InspectionTypeCode
          AND T.IsActive=1
          AND V.Version=@p_TemplateVersion
          AND (V.Status='Active' OR
          (V.Status='Archieved' AND EXISTS
             (SELECT 1 FROM dbo.PIP_T_Inspection Existing
              WHERE Existing.ClientInspectionID=@p_ClientInspectionID
                AND Existing.IDTemplate=T.IDTemplate AND Existing.IDVersion=V.IDVersion
                AND Existing.OwnerEmployeeID=@p_UpdateUser AND ISNULL(Existing.IsApprove,0)=0)))
          AND
          (
              ISNULL(T.EmployeeList,'')='' OR (';' + ISNULL(T.EmployeeList,'') + ';' LIKE '%;' + @p_UpdateUser + ';%')
          )
          AND
          (
              ISNULL(T.CompanyScope,'S')='A'
              OR (';' + ISNULL(T.CompanyLocationList, '') + ';' LIKE '%;' + @p_CompanyID + @p_LocationID + ';%')
          )
    )
    BEGIN
        RAISERROR('Inspection template is inactive, missing, or unavailable for this company/location.', 16, 1);
        RETURN;
    END;

    /* EQueryP may already own a transaction. Use a savepoint when nested so this
       procedure never rolls back the caller's transaction on a handled conflict. */
    DECLARE @V_EntryTranCount INT;
    SET @V_EntryTranCount = @@TRANCOUNT;

    BEGIN TRY
        IF @V_EntryTranCount = 0
            BEGIN TRANSACTION;
        ELSE
            SAVE TRANSACTION InspectionSyncSave;

        /* One scheduled inspection per company/location, section, date, type, detail, time, and template.
           ClientInspectionID is excluded so syncing an existing offline record remains an update. */
        IF EXISTS
        (
            SELECT 1
            FROM dbo.PIP_T_Inspection WITH (UPDLOCK, HOLDLOCK)
            WHERE ClientInspectionID <> @p_ClientInspectionID
              AND ISNULL(IsActive,1)=1
              AND CompanyID = @p_CompanyID
              AND LocationID = @p_LocationID
              AND SectionID = @p_SectionID
              AND InspectionTypeCode = @p_InspectionTypeCode
              AND InspectionTypeDetailCode = @p_InspectionTypeDetailCode
              AND ISNULL(ScheduledTime, '') = ISNULL(@p_ScheduledTime, '')
              AND IDTemplate = @p_IDTemplate
              AND InspectionDate >= CONVERT(DATETIME, CONVERT(VARCHAR(10), @p_InspectionDate, 120), 120)
              AND InspectionDate < DATEADD(DAY, 1, CONVERT(DATETIME, CONVERT(VARCHAR(10), @p_InspectionDate, 120), 120))
        )
        BEGIN
            RAISERROR('An inspection already exists for this date, company/location, section, type, detail, schedule, and template.', 16, 1);
        END;

        DECLARE @V_InspectionID BIGINT;
        DECLARE @V_InspectionTypeName VARCHAR(100);
        DECLARE @V_InspectionTypeDetailName VARCHAR(100);
        DECLARE @V_TemplateName VARCHAR(200);

        SELECT @V_InspectionTypeName=T.InspectionTypeName,
               @V_InspectionTypeDetailName=D.InspectionTypeDetailName
        FROM dbo.PIP_M_InspectionType T
        JOIN dbo.PIP_M_InspectionTypeDetail D ON D.InspectionTypeCode=T.InspectionTypeCode
        WHERE T.InspectionTypeCode=@p_InspectionTypeCode
          AND D.InspectionTypeDetailCode=@p_InspectionTypeDetailCode
          AND D.CompanyID=@p_CompanyID
          AND D.LocationID=@p_LocationID;

        SELECT @V_TemplateName=TemplateName
        FROM dbo.PIP_M_CheckListWOPM
        WHERE IDTemplate=@p_IDTemplate;

        SELECT @V_InspectionID=InspectionID
        FROM dbo.PIP_T_Inspection WITH (UPDLOCK, HOLDLOCK)
        WHERE ClientInspectionID=@p_ClientInspectionID;

        IF @V_InspectionID IS NOT NULL
        BEGIN
            IF NOT EXISTS
            (
                SELECT 1 FROM dbo.PIP_T_Inspection
                WHERE InspectionID=@V_InspectionID AND OwnerEmployeeID=@p_UpdateUser
            )
            BEGIN
                RAISERROR('This offline inspection belongs to another employee.', 16, 1);
            END;

            IF EXISTS
            (
                SELECT 1 FROM dbo.PIP_T_Inspection
                WHERE InspectionID=@V_InspectionID AND ISNULL(IsActive,1)=0
            )
            BEGIN
                RAISERROR('Inactive inspections cannot be changed or resubmitted.', 16, 1);
            END;

            IF EXISTS
            (
                SELECT 1 FROM dbo.PIP_T_Inspection
                WHERE InspectionID=@V_InspectionID AND ISNULL(IsApprove,0)=1
            )
            BEGIN
                RAISERROR('Approved inspections are locked and cannot be changed.', 16, 1);
            END;

            UPDATE dbo.PIP_T_Inspection
            SET InspectionNo=CASE WHEN ISNULL(@p_InspectionNo,'')<>'' THEN @p_InspectionNo ELSE InspectionNo END,
                CompanyID=@p_CompanyID, LocationID=@p_LocationID, SectionID=@p_SectionID,
                InspectionTypeCode=@p_InspectionTypeCode, InspectionTypeName=@V_InspectionTypeName,
                InspectionTypeDetailCode=@p_InspectionTypeDetailCode,
                InspectionTypeDetailName=@V_InspectionTypeDetailName, ScheduledTime=@p_ScheduledTime,
                IDTemplate=@p_IDTemplate, IDVersion=@p_IDVersion, TemplateNo=@p_TemplateNo,
                TemplateName=@V_TemplateName, TemplateVersion=@p_TemplateVersion,
                InspectionDate=@p_InspectionDate, StartDateTime=@p_StartDateTime,
                EndDateTime=@p_EndDateTime, Status=@p_Status, Summary=@p_Summary,
                FollowUp=@p_FollowUp, FollowUpSection=ISNULL(@p_FollowUpSection,''), FollowUpRemark=ISNULL(@p_FollowUpRemark,''),
                IsApprove=0, ApprovedBy='', ApprovedDate=CONVERT(DATETIME,'19000101',112), ApprovalRemark='',
                ReasonVoid='',
                UpdateUser=@p_UpdateUser, UpdateDate=GETDATE(), SyncDate=GETDATE()
            WHERE InspectionID=@V_InspectionID;
        END
        ELSE
        BEGIN
            DECLARE @V_InspectionNo VARCHAR(10);
            DECLARE @V_AutoNum VARCHAR(30);
            DECLARE @V_AutoNumResult TABLE (ID VARCHAR(10), SeqNo INT);

            SET @V_InspectionNo = NULLIF(LTRIM(RTRIM(@p_InspectionNo)), '');
            IF @V_InspectionNo IS NULL
            BEGIN
                SET @V_AutoNum = 'PIPINS' + @p_CompanyID + @p_LocationID;
                INSERT INTO @V_AutoNumResult
                EXECUTE dbo.sp_GetGenerateAutoNumID 'pip', @V_AutoNum, 1;
                SELECT @V_InspectionNo = ID FROM @V_AutoNumResult;
            END;

            IF ISNULL(@V_InspectionNo, '') = ''
            BEGIN
                RAISERROR('Failed to generate InspectionNo.', 16, 1);
            END;

            SELECT @V_InspectionID=ISNULL(MAX(InspectionID),0)+1
            FROM dbo.PIP_T_Inspection WITH (UPDLOCK, HOLDLOCK);

            INSERT dbo.PIP_T_Inspection
            (
                InspectionID, ClientInspectionID, InspectionNo, CompanyID, LocationID, SectionID,
                InspectionTypeCode, InspectionTypeName, InspectionTypeDetailCode,
                InspectionTypeDetailName, ScheduledTime, IDTemplate, IDVersion,
                TemplateNo, TemplateName, TemplateVersion, InspectionDate,
                StartDateTime, EndDateTime, Status, Summary, FollowUp, FollowUpSection, FollowUpRemark, OwnerEmployeeID,
                EntryUser, EntryDate, UpdateUser, UpdateDate, SyncDate, IsActive
            )
            VALUES
            (
                @V_InspectionID, @p_ClientInspectionID, @V_InspectionNo, @p_CompanyID, @p_LocationID, @p_SectionID,
                @p_InspectionTypeCode, @V_InspectionTypeName, @p_InspectionTypeDetailCode,
                @V_InspectionTypeDetailName, @p_ScheduledTime, @p_IDTemplate, @p_IDVersion,
                @p_TemplateNo, @V_TemplateName, @p_TemplateVersion, @p_InspectionDate,
                @p_StartDateTime, @p_EndDateTime, @p_Status, @p_Summary, @p_FollowUp,
                ISNULL(@p_FollowUpSection,''), ISNULL(@p_FollowUpRemark,''), @p_UpdateUser,
                @p_UpdateUser, GETDATE(), @p_UpdateUser, GETDATE(), GETDATE(), 1
            );
        END;

        DELETE V
        FROM dbo.PIP_T_InspectionResultValue V
        JOIN dbo.PIP_T_InspectionResult R ON R.InspectionResultID=V.InspectionResultID
        WHERE R.InspectionID=@V_InspectionID;

        DELETE FROM dbo.PIP_T_InspectionResult WHERE InspectionID=@V_InspectionID;

        DECLARE @V_ResultMap TABLE (IDItem INT NOT NULL, InspectionResultID BIGINT NOT NULL);
        DECLARE @V_NextResultID BIGINT;
        SELECT @V_NextResultID=ISNULL(MAX(InspectionResultID),0)
        FROM dbo.PIP_T_InspectionResult WITH (UPDLOCK, HOLDLOCK);
        ;WITH ResultRows AS
        (
            SELECT I.*,
                   @V_NextResultID + ROW_NUMBER() OVER (ORDER BY I.IDItem) AS NewInspectionResultID
            FROM @tbl_Items I
        )
        INSERT dbo.PIP_T_InspectionResult
        (
            InspectionResultID, InspectionID, IDItem, ItemLabel, ItemType, Value, Remark1, Remark2, IsAnomali,
            DisplayOrder, EntryUser, EntryDate, UpdateUser, UpdateDate
        )
        OUTPUT inserted.IDItem, inserted.InspectionResultID
            INTO @V_ResultMap (IDItem, InspectionResultID)
        SELECT I.NewInspectionResultID, @V_InspectionID, I.IDItem, I.ItemLabel, I.ItemType, I.Value, I.Remark1, I.Remark2,
               I.IsAnomali, I.DisplayOrder, @p_UpdateUser, GETDATE(), @p_UpdateUser, GETDATE()
        FROM ResultRows I;

        INSERT dbo.PIP_T_InspectionResultValue
        (
            InspectionResultID, ObjectIndex, FieldName, Value, EntryUser, EntryDate, UpdateUser, UpdateDate
        )
        SELECT M.InspectionResultID, V.ObjectIndex, V.FieldName, V.Value,
               @p_UpdateUser, GETDATE(), @p_UpdateUser, GETDATE()
        FROM @tbl_ItemValues V
        JOIN @V_ResultMap M ON M.IDItem=V.IDItem;

        /* Header, checklist, and photo metadata commit or roll back together. */
        DELETE FROM dbo.PIP_T_InspectionPhoto WHERE InspectionID=@V_InspectionID;
        DECLARE @V_NextPhotoID BIGINT;
        SELECT @V_NextPhotoID=ISNULL(MAX(InspectionPhotoID),0)
        FROM dbo.PIP_T_InspectionPhoto WITH (UPDLOCK, HOLDLOCK);
        ;WITH PhotoRows AS
        (
            SELECT P.C1, P.C2, P.C3, P.C4,
                   @V_NextPhotoID + ROW_NUMBER() OVER (ORDER BY CAST(ISNULL(NULLIF(P.C4,''),'0') AS INT), P.C2) AS NewInspectionPhotoID
            FROM @p_Photos P
        )
        INSERT dbo.PIP_T_InspectionPhoto
            (InspectionPhotoID, InspectionID, PhotoType, FileName, FileType, SeqNo, EntryUser, EntryDate)
        SELECT P.NewInspectionPhotoID, @V_InspectionID, ISNULL(P.C1,''), ISNULL(P.C2,''), ISNULL(P.C3,''),
               CAST(ISNULL(NULLIF(P.C4,''),'0') AS INT), @p_UpdateUser, GETDATE()
        FROM PhotoRows P;

        IF @V_EntryTranCount = 0
            COMMIT TRANSACTION;

        SELECT @V_InspectionID AS InspectionID,
               @p_ClientInspectionID AS ClientInspectionID,
               InspectionNo
        FROM dbo.PIP_T_Inspection
        WHERE InspectionID=@V_InspectionID;
    END TRY
    BEGIN CATCH
        IF XACT_STATE() = 1
        BEGIN
            IF @V_EntryTranCount = 0
                ROLLBACK TRANSACTION;
            ELSE
                ROLLBACK TRANSACTION InspectionSyncSave;
        END
        ELSE IF XACT_STATE() = -1 AND @V_EntryTranCount = 0
            ROLLBACK TRANSACTION;

        DECLARE @V_ErrorMessage NVARCHAR(4000);
        DECLARE @V_ErrorSeverity INT;
        DECLARE @V_ErrorState INT;
        SELECT @V_ErrorMessage=ERROR_MESSAGE(), @V_ErrorSeverity=ERROR_SEVERITY(), @V_ErrorState=ERROR_STATE();
        RAISERROR(@V_ErrorMessage, @V_ErrorSeverity, @V_ErrorState);
    END CATCH;
END;
